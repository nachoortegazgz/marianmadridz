    if (typeof value === "number") return Number.isFinite(value) ? value : null;
    let raw = String(value ?? "").trim().replace(/\s+/g, "").replace(/€$/i, "");
    if (!raw) return null;

    if (sign) raw = raw.slice(1);

    const comma = raw.lastIndexOf(",");
    const dot = raw.lastIndexOf(".");
    let integerPart = raw;

      const decimalSeparator = comma > dot ? "," : ".";
      const groupSeparator = decimalSeparator === "," ? "." : ",";
      const parts = raw.split(decimalSeparator);
      if (parts.length !== 2 || !/^\d{1,2}$/.test(parts[1])) return null;
      const groups = parts[0].split(groupSeparator);

  /* ---------- Envío al controlador (postMessage) ---------- */
  function call(type, payload, responseType) {
    return new Promise((resolve, reject) => {
      const id = "req-" + crypto.randomUUID().replace(/-/g, "");
      let timer = null;
      const cleanup = () => {
        window.removeEventListener("message", handler);
        if (timer) clearTimeout(timer);
      };
      const handler = (event) => {
        if (event.source !== window.parent || event.origin !== PARENT_ORIGIN) return;
        const data = event.data || {};
        if (data.type === responseType && (data.messageId === id || data.requestId === id)) {
          cleanup();
          const response = data.payload || {};
          if (data.error || response.status === "ERROR") {
            const sourceError = response.error || {};
            const error = new Error(sourceError.message || sourceError.code || data.error || "TPV_OPERATION_FAILED");
            error.code = sourceError.code || data.error || "TPV_OPERATION_FAILED";
            error.retryable = sourceError.retryable === true;
            error.requiresReconciliation = sourceError.requiresReconciliation === true;
            error.recoverable = sourceError.recoverable === true;
            if (sourceError.outcomeUnknown === true) error.outcomeUnknown = true;
            if (sourceError.operationMayContinue === true) error.operationMayContinue = true;
            reject(error);
          } else if (response.status === "PENDING") {
            const error = new Error(response.reason || "TPV_OPERATION_PENDING");
            error.code = response.reason || "TPV_OPERATION_PENDING";
            error.status = "PENDING";
            error.retryable = false;
            error.requiresReconciliation = response.requiresReconciliation === true;
            reject(error);
          } else {
            resolve(response.data !== undefined ? response.data : response);
          }
        }
      };
      window.addEventListener("message", handler);
      timer = setTimeout(() => {
        cleanup();
        const error = new Error("TPV_TIMEOUT");
        error.code = "TPV_TIMEOUT";
        error.outcomeUnknown = true;
        error.operationMayContinue = true;
        error.retryable = false;
        error.requiresReconciliation = true;
        reject(error);
      }, 15000);
      try { postToParent({ type, payload, messageId: id, version: 1 }); }
      catch (originError) { cleanup(); reject(originError); }
    });
  }

  /* ---------- Idempotencia local ---------- */
  function keyedAction(keyPrefix, type, payload, responseType) {
    const storageKey = "tpv_key_" + keyPrefix;
    let key = sessionStorage.getItem(storageKey);
    if (!key) {
      key = keyPrefix + "_" + crypto.randomUUID().replace(/-/g, "");
      sessionStorage.setItem(storageKey, key);
    }
    const fullPayload = Object.assign({}, payload, { idempotencyKey: key });
    return call(type, fullPayload, responseType).then((result) => {
      if (result !== null && result !== undefined) {
        sessionStorage.removeItem(storageKey);
      }
      return result;
    });
  }

    if (state.busy) return;

      return result;

      setBusy(false);
    }
  }

  /* ---------- Catálogo nativo: Bookings y Stores Catalog V3 ---------- */
  function addCatalogLine(item, priceMinor, varianteId = null, variantLabel = "") {
    if (!["PRODUCTO", "SERVICIO"].includes(String(item?.tipoLinea || "").toUpperCase())) {
      toast("Solo se admiten líneas del catálogo con clasificación fiscal Wix.");
      return;
    }
    if (!Number.isSafeInteger(priceMinor) || priceMinor <= 0) {
      toast("El artículo no tiene un precio válido.");
      return;
    }
    state.items.push({
      id: item.itemId,
      tipoLinea: String(item.tipoLinea).toUpperCase(),
      name: [item.concepto, variantLabel].filter(Boolean).join(" · "),
      qty: 1,
      varianteId,
    });
    state.payments = [];
    renderTicket();
    refreshSaleQuote();
  }

  async function selectCatalogItem(item) {
    if (item.tipoLinea === "PRODUCTO") {
      const result = await call("CATALOG_VARIANTS", { productId: item.itemId }, "CATALOG_VARIANTS_RES");
      const variants = (result?.variantes || []).filter((variant) =>
        variant.disponible === true && Number.isSafeInteger(variant.precioMinor) &&
        typeof variant.varianteId === "string" && variant.varianteId.length > 0
      );
      if (!variants.length) {
        toast("No hay variantes disponibles con precio válido.");
        return;
      }
      if (variants.length === 1) {
        const selected = variants[0];
        addCatalogLine(item, selected.precioMinor, selected.varianteId, selected.concepto);
        return;
      }
      openModal("Seleccionar variante", `
        <select id="variantChoice" class="field">
          ${variants.map((variant) => `<option value="${esc(variant.varianteId)}">${esc(variant.concepto)} · ${money(variant.precioMinor / 100)}</option>`).join("")}
        </select>
      `, async () => {
        const selected = variants.find((variant) => variant.varianteId === $("variantChoice").value);
        if (!selected) throw new Error("TPV_VARIANT_REQUIRED");
        addCatalogLine(item, selected.precioMinor, selected.varianteId, selected.concepto);
      });
      return;
    }
    if (!Number.isSafeInteger(item.precioMinor)) {
      toast("El artículo no tiene un precio fijo configurado.");
      return;
    }
    addCatalogLine(item, item.precioMinor);
  }

  /* ---------- Preparación fiscal autoritativa ---------- */
  function saleLineRequests() {
    return state.items.map((item) => ({
      tipoLinea: item.tipoLinea,
      itemId: item.id,
      ...(item.varianteId ? { varianteId: item.varianteId } : {}),
      cantidad: item.qty,
      profesionalId: state.profesionalId || null,
      descuentoImporte: 0,
    }));
  }

  async function requestSaleQuote(payments = []) {
    return call("SALE_VALIDATE", {
      lineItems: saleLineRequests(),
      payments: payments.map((payment) => ({
        method: payment.method,
        amount: payment.amountMinor / 100,
        change: payment.changeMinor / 100,
      })),
    }, "SALE_VALIDATE_RES");
  }

  async function refreshSaleQuote() {
    const sequence = ++state.quoteSequence;
    if (!state.items.length) {
      state.saleQuote = null;
      state.quoteLoading = false;
      state.quoteError = null;
      renderTicket();
      return;
    }

    state.saleQuote = null;
    state.quoteLoading = true;
    state.quoteError = null;
    renderTicket();

    try {
      const quote = await requestSaleQuote(state.payments);
      if (sequence !== state.quoteSequence) return;
      state.saleQuote = quote;
    } catch (err) {
      if (sequence !== state.quoteSequence) return;
      state.quoteError = err.code || err.message || "TPV_SALE_PREPARATION_FAILED";
    } finally {
      if (sequence === state.quoteSequence) {
        state.quoteLoading = false;
        renderTicket();
      }
    }
  }

  /* ---------- Ticket ---------- */
  function renderTicket() {
    const container = $("ticketLines");
    const quoteLines = Array.isArray(state.saleQuote?.lineas) ? state.saleQuote.lineas : [];
    if (!state.items.length && state.lastSale) {
      const sale = state.lastSale;
      const methods = (sale.pagos || []).map((payment) => payment.method).filter(Boolean).join(", ");
      const hasProducts = (sale.lineas || []).some((line) => line.tipoLinea === "PRODUCTO");
      container.innerHTML = `
        <div class="ticket-line"><strong>VENTA</strong></div>
        <div class="ticket-line"><span>Referencia</span><strong>${esc(sale.saleId || "")}</strong></div>
        <div class="ticket-line"><span>Total simulado</span><strong>${money(sale.totalVenta || 0)}</strong></div>
        <div class="ticket-line"><span>Medio declarado</span><strong>${esc(methods || "—")}</strong></div>
        ${hasProducts ? '<div class="ticket-line"><span>Stock</span><strong>No consultado ni modificado</strong></div>' : ''}
      `;
    } else if (!state.items.length) {
      container.innerHTML = '<p style="color:var(--muted);">Añade conceptos del catálogo.</p>';
    } else {
      container.innerHTML = state.items.map((item, i) => {
        const lineTotal = quoteLines[i]?.totalLinea;
        const displayName = quoteLines[i]?.concepto || item.name;
        const displayTotal = lineTotal !== undefined
          ? money(lineTotal)
          : (state.quoteLoading ? "Calculando…" : "No disponible");
        return `
          <div class="ticket-line">
            <span>${esc(displayName)}</span>
            <span class="qty">
              <button data-action="minus" data-idx="${i}">−</button>
              <span>${item.qty}</span>
              <button data-action="plus" data-idx="${i}">+</button>
            </span>
            <span>${displayTotal}</span>
          </div>
        `;
      }).join("");
      container.querySelectorAll("button[data-action]").forEach((btn) => {
        btn.onclick = () => {
          const idx = Number(btn.dataset.idx);
          if (btn.dataset.action === "plus") state.items[idx].qty++;
          else if (btn.dataset.action === "minus") {
            state.items[idx].qty--;
            if (state.items[idx].qty <= 0) state.items.splice(idx, 1);
          }
          state.payments = [];
          renderTicket();
          refreshSaleQuote();
        };
      });
    }

    const quote = state.saleQuote;
    const hasQuote = quote && Number.isFinite(Number(quote.totalVenta));
    $("subtotal").textContent = hasQuote ? money(quote.subtotalNeto) : (state.items.length ? (state.quoteLoading ? "Calculando…" : "—") : money(0));
    $("tax").value = hasQuote ? money(quote.impuestosTotal) : (state.items.length ? (state.quoteLoading ? "Calculando…" : "—") : money(0));
    $("total").textContent = hasQuote ? money(quote.totalVenta) : (state.items.length ? (state.quoteLoading ? "Calculando…" : "—") : money(0));

    const readiness = $("saleReadiness");
    if (hasQuote) {
      const blockers = quote.saleReadinessBlockers || [];
      const messages = [];
            if (blockers.includes("WIX_TAX_REGION_EXCLUDES_TAX")) messages.push("Wix trata los precios del catálogo como importes sin IVA; se calcula el impuesto por separado.");
      if (blockers.includes("FISCAL_CONFIG_UNVERIFIED")) messages.push("Configuración fiscal no verificada.");
      if (blockers.includes("FISCAL_WRITER_DISABLED")) messages.push("Registro fiscal deshabilitado.");
      if (blockers.includes("PAYMENT_NOT_VERIFIED")) messages.push("Pago no verificado.");
      if (blockers.includes("TPV_WRITE_V1_TAX_RATE_UNVERIFIABLE")) messages.push("La tasa impositiva de alguna línea no puede verificarse.");
      if (blockers.includes("TPV_WRITE_V1_MULTI_VAT_UNSUPPORTED")) messages.push("La venta combina varias tasas de IVA; la escritura V1 solo admite una.");
      if (blockers.includes("TPV_WRITE_V1_PAYMENT_REQUIRED")) messages.push("Añade un medio de pago.");
      if (blockers.includes("TPV_WRITE_V1_SPLIT_TENDER_UNSUPPORTED")) messages.push("La escritura V1 admite un único medio de pago por venta.");
      if (blockers.includes("TPV_WRITE_V1_PAYMENT_METHOD_UNSUPPORTED")) messages.push("El medio de pago no está habilitado en el contrato V1.");
      if (blockers.includes("TPV_WRITE_V1_PAYMENT_TOTAL_MISMATCH")) messages.push("El importe aplicado del pago no coincide con el total de la venta.");
      if (blockers.includes("INVENTORY_POLICY_NOT_CONFIGURED")) messages.push("Inventario de productos no configurado en la ubicación física del salón.");
      if (blockers.includes("CONSUMER_PRICE_PRESENTATION_UNVERIFIED")) messages.push("Presentación del precio final al consumidor pendiente de validación.");
      readiness.textContent = quote.commercialRecordReady === true && state.operationalWritesAvailable === true
        ? "Registro comercial disponible; se guardará pendiente de cobro. " + messages.join(" ")
        : "No se puede registrar la venta: " + (messages.join(" ") || "faltan requisitos de catálogo o sesión.");
    } else if (state.quoteLoading) {
      readiness.textContent = "Consultando catálogo y cálculo fiscal de Wix…";
    } else if (state.quoteError) {
      readiness.textContent = "No se pudo preparar la venta: " + publicErrorMessage(state.quoteError);
    } else {
      readiness.textContent = "Añade líneas para calcular el total fiscal.";
    }
    renderPaymentSummary();
  }

  /* ---------- Pagos ---------- */
  function paymentTotals() {
    const totalsMinor = Object.fromEntries(PAYMENT_METHODS.map((method) => [method, 0]));
    state.payments.forEach((payment) => {
      totalsMinor[payment.method] += Math.max(0, payment.amountMinor - payment.changeMinor);
    });
    return Object.fromEntries(
      PAYMENT_METHODS.map((method) => [method, totalsMinor[method] / 100])
    );
  }

  function renderPaymentSummary() {
    const totalMinor = state.saleQuote && Number.isFinite(Number(state.saleQuote.totalVenta))
      ? Math.round(Number(state.saleQuote.totalVenta) * 100)
      : null;
    const appliedMinor = state.payments.reduce((sum, payment) =>
      sum + payment.amountMinor - payment.changeMinor, 0);
    const pendingMinor = totalMinor === null ? null : Math.max(0, totalMinor - appliedMinor);
    const changeMinor = state.payments.reduce((sum, payment) => sum + payment.changeMinor, 0);
    const paymentReady = state.saleQuote?.catalogVerified === true && state.saleQuote?.taxVerified === true && state.saleQuote?.saleReady === true && state.operationalWritesAvailable === true;
    $("paymentAmount").disabled = !paymentReady;
    $("addPayment").disabled = !paymentReady;
    document.querySelectorAll("[data-cash]").forEach((button) => { button.disabled = !paymentReady; });
    $("paymentMethods").querySelectorAll("button[data-method]").forEach((button) => {
      const unsupportedTender = ["BONO", "TARJETA_REGALO", "OTROS"].includes(button.dataset.method);
      button.disabled = !paymentReady || unsupportedTender;
      if (unsupportedTender) button.title = "Pendiente de validación de saldo en backend.";
    });
    $("applied").textContent = money(appliedMinor / 100);
    $("pending").textContent = pendingMinor === null ? (state.quoteLoading ? "Calculando…" : "—") : money(pendingMinor / 100);
    $("change").textContent = money(changeMinor / 100);

    const totals = paymentTotals();
    const rows = state.payments.map((payment, index) => `
      <div class="payment-row" style="display:flex;align-items:center;gap:8px;padding:6px 0;">

  function csvCell(value) {
    let text = value === null || value === undefined
      ? ""
      : (typeof value === "object" ? JSON.stringify(value) : String(value));
    if (/^[=+\-@\t\r]/.test(text)) text = "'" + text;
    return "\"" + text.replaceAll("\"", "\"\"") + "\"";
  }

  function downloadCsv(rows, filename, columns) {
    const csv = [
      columns.map(csvCell).join(";"),
      ...rows.map((row) => columns.map((column) => csvCell(row[column])).join(";")),
    ].join("\n");
    const blob = new Blob(["\ufeff" + csv], { type: "text/csv;charset=utf-8" });
    const url = URL.createObjectURL(blob);
    const link = document.createElement("a");
    link.href = url;
    link.download = filename;
    link.style.display = "none";
    document.body.appendChild(link);
    link.click();
    link.remove();
    setTimeout(() => URL.revokeObjectURL(url), 1000);
  }

  async function exportData(operationType, filename, columns) {
    const sessionId = state.session?.sessionId;
    if (!sessionId) throw new Error("No hay una sesión de caja seleccionada.");

    if (result?.truncated === true) throw new Error("La exportación está incompleta; no se ha generado el CSV.");
    const rows = Array.isArray(result?.items) ? result.items : [];
    if (!rows.length) {
      toast("No hay datos para exportar.");
      return;
    }
    downloadCsv(rows, filename, columns);
    toast("Exportación CSV generada: " + rows.length + " registros.");
  }

  /* ---------- Handshake y contexto del controlador Wix ---------- */
  const BRIDGE_PROTOCOL_VERSION = 1;
  const HANDSHAKE_RETRY_MS = 500;
  // Retry for 60 s; keep a 5 s margin before the 65 s Velo watchdog.
  const HANDSHAKE_MAX_ATTEMPTS = 120;
  let handshakeAttempts = 0;
  let handshakeTimer = null;
  let widgetInitialized = false;

  function initializeWidgetAfterHandshake() {
    if (widgetInitialized) return;
    widgetInitialized = true;
    if (handshakeTimer) {
      clearTimeout(handshakeTimer);
      handshakeTimer = null;
    }

    loadCatalog("SERVICES").catch((e) => error(e.message));
    call("CATALOG_STAFF", {}, "CATALOG_STAFF_RES").then((data) => {
      state.staff = data.items || [];
      renderProfessionals();
    }).catch((e) => error(e.message));
    refreshCashState().catch((e) => error(e.message));
  }

  function sendHandshake() {
    if (widgetInitialized) return;
    if (handshakeAttempts >= HANDSHAKE_MAX_ATTEMPTS) {
      handshakeTimer = null;
      $("connStatus").className = "dot error";
      $("connLabel").textContent = "Sin conexión con Wix";
      error("No se recibió respuesta del controlador TPV.");
      return;
    }

    handshakeAttempts += 1;
    try {
      postToParent({
        type: "MM_READY",
        version: BRIDGE_PROTOCOL_VERSION,
        messageId: "tpv-ready-" + crypto.randomUUID().replace(/-/g, ""),
      });
    } catch (originError) {
      handshakeTimer = null;
      $("connStatus").className = "dot error";
      $("connLabel").textContent = "Origen de página no disponible";
      error(originError.code || "TPV_PARENT_ORIGIN_UNAVAILABLE");
      return;
    }
    handshakeTimer = setTimeout(() => {
      handshakeTimer = null;
      sendHandshake();
    }, HANDSHAKE_RETRY_MS);
  }

  window.addEventListener("message", (event) => {
    if (event.source !== window.parent) return;
    const data = event.data || {};
    if (data.type !== "MM_CONTEXT") return;

    if (handshakeTimer) {
      clearTimeout(handshakeTimer);
      handshakeTimer = null;
    }

    const context = data.payload || {};
    const connected = context.status === "SUCCESS" &&
      context.data?.component === "htmlTPV" &&
      context.data?.protocolVersion === BRIDGE_PROTOCOL_VERSION;
    state.capabilities = context.data?.capabilities &&
      typeof context.data.capabilities === "object"
      ? context.data.capabilities
      : {};
    state.operationalWritesAvailable = context.data?.operationalWritesAvailable === true;
    $("connStatus").className = "dot " + (connected ? "ok" : "error");
    $("connLabel").textContent = connected ? (state.operationalWritesAvailable ? "Conectado · Operaciones autorizadas" : "Conectado · Solo lectura") : "No autorizado";

    if (connected) {
      initializeWidgetAfterHandshake();
    } else {
      error(context.error?.code || "No se pudo validar el contexto del TPV.");
    }
  });

  /* ---------- Inicialización ---------- */
  function init() {
    $("connStatus").className = "dot";
    $("connLabel").textContent = "Conectando…";
    sendHandshake();
  }

  init();
})();
