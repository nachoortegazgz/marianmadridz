<script>
let ticket = [];
let caja = {
  ventas: 0,
  efectivo: 0,
  tarjeta: 0,
  bizum: 0,
  otros: 0,
  cajon: 0
};

function euro(value) {
  return value.toLocaleString("es-ES", {
    style: "currency",
    currency: "EUR"
  });
}

function addProduct(name, price) {
  const existing = ticket.find(item => item.name === name);

  if (existing) {
    existing.quantity++;
  } else {
    ticket.push({
      name,
      price,
      quantity: 1
    });
  }

  renderTicket();
  showMessage(name + " añadido al ticket");
}

function changeQuantity(index, amount) {
  ticket[index].quantity += amount;

  if (ticket[index].quantity <= 0) {
    ticket.splice(index, 1);
  }

  renderTicket();
}

function removeItem(index) {
  ticket.splice(index, 1);
  renderTicket();
}

function getTotal() {
  return ticket.reduce((sum, item) => {
    return sum + item.price * item.quantity;
  }, 0);
}

function renderTicket() {
  const container = document.getElementById("ticketItems");

  if (ticket.length === 0) {
    container.innerHTML = `
      <div class="ticket-empty">
        Selecciona un producto o servicio para comenzar.
      </div>
    `;
  } else {
    container.innerHTML = ticket.map((item, index) => `
      <div class="ticket-item">
        <div>
          <div class="ticket-item-title">${item.name}</div>
          <div class="ticket-item-details">
            ${euro(item.price)} · ${euro(item.price * item.quantity)}
          </div>
        </div>

        <div class="item-actions">
          <div class="quantity">
            <button onclick="changeQuantity(${index}, -1)">−</button>
            <strong>${item.quantity}</strong>
            <button onclick="changeQuantity(${index}, 1)">+</button>
          </div>

          <button class="remove" onclick="removeItem(${index})">×</button>
        </div>
      </div>
    `).join("");
  }

  const total = getTotal();

  document.getElementById("subtotal").textContent = euro(total);
  document.getElementById("total").textContent = euro(total);
}

function clearTicket() {
  if (ticket.length === 0) {
    showMessage("No hay ningún ticket para anular");
    return;
  }

  ticket = [];
  renderTicket();
  showMessage("Ticket anulado");
}

function pay(method) {
  const total = getTotal();

  if (total <= 0) {
    showMessage("Añade al menos un producto o servicio");
    return;
  }

  caja.ventas += total;

  if (method === "Efectivo") {
    caja.efectivo += total;
    caja.cajon += total;
  } else if (method === "Tarjeta") {
    caja.tarjeta += total;
  } else if (method === "Bizum") {
    caja.bizum += total;
  } else {
    caja.otros += total;
  }

  ticket = [];
  renderTicket();
  renderCaja();

  showMessage("Cobro registrado: " + method + " · " + euro(total));
}

function renderCaja() {
  document.getElementById("salesTotal").textContent = euro(caja.ventas);
  document.getElementById("cashTotal").textContent = euro(caja.efectivo);
  document.getElementById("cardTotal").textContent = euro(caja.tarjeta);
  document.getElementById("bizumTotal").textContent = euro(caja.bizum);
  document.getElementById("otherTotal").textContent = euro(caja.otros);
  document.getElementById("drawerTotal").textContent = euro(caja.cajon);
}

function filterProducts() {
  const search = document
    .getElementById("search")
    .value
    .toLowerCase();

  document.querySelectorAll(".product").forEach(product => {
    const name = product.dataset.name.toLowerCase();
    product.style.display = name.includes(search) ? "block" : "none";
  });
}

function showMessage(message) {
  const toast = document.getElementById("toast");
  toast.textContent = message;
  toast.classList.add("show");

  setTimeout(() => {
    toast.classList.remove("show");
  }, 2500);
}

function updateClock() {
  const now = new Date();

  document.getElementById("clock").textContent =
    now.toLocaleTimeString("es-ES", {
      hour: "2-digit",
      minute: "2-digit"
    });
}

setInterval(updateClock, 1000);
updateClock();
renderTicket();
renderCaja();
</script>
