  function normalizeError(sourceError, fallbackCode = "TPV_OPERATION_FAILED") {
    if (sourceError && typeof sourceError === "object") {
      return {
        code: String(sourceError.code || fallbackCode),
        message: sourceError.message || null,
        userMessage: sourceError.userMessage || null,
        technicalDetail: sourceError.technicalDetail || null,
        remediation: Array.isArray(sourceError.remediation) ? sourceError.remediation.filter(Boolean) : [],
        retryable: sourceError.retryable === true,
        recoverable: sourceError.recoverable === true,
        requiresReconciliation: sourceError.requiresReconciliation === true,
        outcomeUnknown: sourceError.outcomeUnknown === true,
        operationMayContinue: sourceError.operationMayContinue === true,
        status: sourceError.status || null
      };
    }
    return { code: String(sourceError || fallbackCode), message: null, userMessage: null, technicalDetail: null, remediation: [], retryable: false, recoverable: false, requiresReconciliation: false, outcomeUnknown: false, operationMayContinue: false, status: null };
  }
  function error(sourceError) {
    const e = normalizeError(sourceError);
    let message = publicErrorMessage(e);
    if (e.status === "PENDING" || e.outcomeUnknown || e.operationMayContinue) message += " El resultado puede estar pendiente. Consulta el estado antes de repetirla.";
    if (e.remediation.length) message += " " + e.remediation.join(" ");
    toast(message);
  }

  /* ---------- Envío al controlador (postMessage) ---------- */
  function call(type, payload, responseType) {
    return new Promise((resolve, reject) => {
      const id = "req-" + crypto.randomUUID().replace(/-/g, "");
      let timer = null;
      const cleanup = () => { window.removeEventListener("message", handler); if (timer) clearTimeout(timer); };
      const rejectStructured = (source, fallbackCode) => {
        const normalized = normalizeError(source, fallbackCode);
        const operationError = new Error(normalized.userMessage || normalized.message || publicErrorMessage(normalized));
        Object.assign(operationError, normalized);
        reject(operationError);
      };
      const handler = (event) => {
        if (event.source !== window.parent) return;
        // HtmlComponent Preview may deliver an opaque origin. The Wix
        // component boundary plus backend authorization remain the trust
        // boundary; normal browser origins still require the resolved origin.
        if (event.origin && event.origin !== "null" && event.origin !== PARENT_ORIGIN) return;
        const data = event.data || {};
        if (data.type !== responseType || (data.messageId !== id && data.requestId !== id)) return;
        cleanup();
        const response = data.payload || {};
        if (data.error || response.error || response.status === "ERROR") {
          rejectStructured(response.error || data.error || response, "TPV_OPERATION_FAILED");
        } else if (response.status === "PENDING") {
          // A persisted pending result is an operation outcome, not a transport failure.
          // Resolve it so the caller can show the status and release its idempotency key.
          resolve(response);
        } else {
          resolve(response.data !== undefined ? response.data : response);
        }
      };
      window.addEventListener("message", handler);
      timer = setTimeout(() => { cleanup(); rejectStructured({ code: "TPV_TIMEOUT", outcomeUnknown: true, operationMayContinue: true, retryable: false, requiresReconciliation: true }, "TPV_TIMEOUT"); }, 15000);
      try { postToParent({ type, payload, messageId: id, version: 1 }); }
      catch (originError) { cleanup(); rejectStructured(originError, "TPV_PARENT_ORIGIN_UNAVAILABLE"); }
    });
  }

  /* ---------- Idempotencia local ---------- */
  async function keyedAction(keyPrefix, type, payload, responseType) {
    const storageKey = "tpv_key_" + keyPrefix;
    const fingerprintKey = storageKey + "_fingerprint";
    const fingerprint = JSON.stringify(payload);
    let key = sessionStorage.getItem(storageKey);

    // After an unknown timeout, never reuse the old sale key for a changed cart.
    // Read back the old operation first; only a proven absence permits a new key.
    if (key && keyPrefix === "SALE" && sessionStorage.getItem(fingerprintKey) !== fingerprint) {
      try {
        const previous = await call("SALE_READBACK", {
          sessionId: payload.sessionId,
          idempotencyKey: key,
        }, "SALE_READBACK_RES");
        if (previous?.recordPersisted === true) {
          const conflict = new Error("TPV_PREVIOUS_SALE_RECONCILIATION_REQUIRED");
          conflict.code = "TPV_PREVIOUS_SALE_RECONCILIATION_REQUIRED";
          conflict.userMessage = "La venta anterior quedó registrada o requiere conciliación. Consulta su estado antes de crear otra venta.";
          throw conflict;
        }
      } catch (readbackError) {
        if (readbackError.code !== "TPV_SALE_NOT_FOUND") throw readbackError;
      }
      sessionStorage.removeItem(storageKey);
      sessionStorage.removeItem(fingerprintKey);
      key = null;
    }

    if (!key) {
      key = keyPrefix + "_" + crypto.randomUUID().replace(/-/g, "");
      sessionStorage.setItem(storageKey, key);
      if (keyPrefix === "SALE") sessionStorage.setItem(fingerprintKey, fingerprint);
    }
    const fullPayload = Object.assign({}, payload, { idempotencyKey: key });
    return call(type, fullPayload, responseType).then((result) => {
      if (result !== null && result !== undefined) {
        sessionStorage.removeItem(storageKey);
        sessionStorage.removeItem(fingerprintKey);
      }
      return result;
    });
  }

  /* ---------- Catálogo nativo: Bookings y Stores Catalog V3 ---------- */
  function addCatalogLine(item, priceMinor, varianteId = null, variantLabel = "") {
    if (!["PRODUCTO", "SERVICIO"].includes(String(item?.tipoLinea || "").toUpperCase())) {
      toast("Solo se admiten líneas del catálogo con clasificación fiscal Wix.");
      return;
    }
    if (!Number.isSafeInteger(priceMinor) || priceMinor <= 0) {
      toast("El artículo no tiene un precio válido.");
      return;
    }
    state.items.push({
      id: item.itemId,
      tipoLinea: String(item.tipoLinea).toUpperCase(),
      name: [item.concepto, variantLabel].filter(Boolean).join(" · "),
      qty: 1,
      varianteId,
    });
    state.payments = [];
    renderTicket();
    refreshSaleQuote();
  }

  async function selectCatalogItem(item) {
    if (item.tipoLinea === "PRODUCTO") {
      const result = await call("CATALOG_VARIANTS", { productId: item.itemId }, "CATALOG_VARIANTS_RES");
      const variants = (result?.variantes || []).filter((variant) =>
        variant.disponible === true && Number.isSafeInteger(variant.precioMinor) &&
        typeof variant.varianteId === "string" && variant.varianteId.length > 0
      );
      if (!variants.length) {
        toast("No hay variantes disponibles con precio válido.");
        return;
      }
      if (variants.length === 1) {
        const selected = variants[0];
        addCatalogLine(item, selected.precioMinor, selected.varianteId, selected.concepto);
        return;
      }
      openModal("Seleccionar variante", `
        <select id="variantChoice" class="field">
          ${variants.map((variant) => `<option value="${esc(variant.varianteId)}">${esc(variant.concepto)} · ${money(variant.precioMinor / 100)}</option>`).join("")}
        </select>
      `, async () => {
        const selected = variants.find((variant) => variant.varianteId === $("variantChoice").value);
        if (!selected) throw new Error("TPV_VARIANT_REQUIRED");
        addCatalogLine(item, selected.precioMinor, selected.varianteId, selected.concepto);
      });
      return;
    }
    if (!Number.isSafeInteger(item.precioMinor)) {
      toast("El artículo no tiene un precio fijo configurado.");
      return;
    }
    addCatalogLine(item, item.precioMinor);
  }

  function renderCatalog(items) {

    if (!Array.isArray(items) || !items.length) {
      list.innerHTML = "<p>No hay elementos disponibles.</p>";
      return;

      const priceMinor = Number.isSafeInteger(item.precioMinor)
        ? item.precioMinor
        : (item.precioFijo !== null && item.precioFijo !== undefined
          ? Math.round(Number(item.precioFijo) * 100)
          : null);
      const priceLabel = item.requiereSeleccionVariante === true
        ? "Seleccionar variante"
        : (Number.isSafeInteger(priceMinor) ? money(priceMinor / 100) : "Precio no configurado");
      return `
        <div class="catalog-item" data-index="${index}">
          <span>${esc(item.concepto || "Artículo")}</span>
          <span class="price">${esc(priceLabel)}</span>
        </div>
      `;
    }).join("");
    list.querySelectorAll(".catalog-item").forEach((element) => {
      element.onclick = () => {
        const item = items[Number(element.dataset.index)];
        selectCatalogItem(item).catch((err) => err.message === "TPV_TIMEOUT"
          ? toast("Tiempo de espera agotado.")
          : error(err));
      };
    });
  }

  async function loadCatalog(kind) {
    state.catalogKind = kind === "PRODUCTS" ? "PRODUCTS" : "SERVICES";
    const type = kind === "PRODUCTS" ? "CATALOG_PRODUCTS" : "CATALOG_SERVICES";
    const responseType = type + "_RES";
    const result = await call(type, { limit: 100 }, responseType);
    renderCatalog(result?.items || []);
  }

  $("tabServicios").onclick = () => loadCatalog("SERVICES").catch((e) => error(e));
  $("tabProductos").onclick = () => loadCatalog("PRODUCTS").catch((e) => error(e));

  /* ---------- Preparación fiscal autoritativa ---------- */
  function saleLineRequests() {
    return state.items.map((item) => ({
      tipoLinea: item.tipoLinea,
      itemId: item.id,
      ...(item.varianteId ? { varianteId: item.varianteId } : {}),
      cantidad: item.qty,
      profesionalId: state.profesionalId || null,
      descuentoImporte: 0,
    }));
  }

  async function requestSaleQuote(payments = []) {
    return call("SALE_VALIDATE", {
      lineItems: saleLineRequests(),
      payments: payments.map((payment) => ({
        method: payment.method,
        amount: payment.amountMinor / 100,
        change: payment.changeMinor / 100,
      })),
    }, "SALE_VALIDATE_RES");
  }

  async function refreshSaleQuote() {
    const sequence = ++state.quoteSequence;
    if (!state.items.length) {
      state.saleQuote = null;
      state.quoteLoading = false;
      state.quoteError = null;
      renderTicket();
      return;
    }

    state.saleQuote = null;
    state.quoteLoading = true;
    state.quoteError = null;
    renderTicket();

    try {
      const quote = await requestSaleQuote(state.payments);
      if (sequence !== state.quoteSequence) return;
      state.saleQuote = quote;
    } catch (err) {
      if (sequence !== state.quoteSequence) return;
      state.quoteError = normalizeError(err, "TPV_SALE_PREPARATION_FAILED");
    } finally {
      if (sequence === state.quoteSequence) {
        state.quoteLoading = false;
        renderTicket();
      }
    }
  }

  /* ---------- Ticket ---------- */
  function renderTicket() {
    const container = $("ticketLines");
    const quoteLines = Array.isArray(state.saleQuote?.lineas) ? state.saleQuote.lineas : [];
    if (!state.items.length && state.lastSale) {
      const sale = state.lastSale;
      const methods = (sale.pagos || []).map((payment) => payment.method).filter(Boolean).join(", ");
      const hasProducts = (sale.lineas || []).some((line) => line.tipoLinea === "PRODUCTO");
      container.innerHTML = `
        <div class="ticket-line"><strong>VENTA</strong></div>
        <div class="ticket-line"><span>Referencia</span><strong>${esc(sale.saleId || "")}</strong></div>
        <div class="ticket-line"><span>Total pendiente de cobro</span><strong>${money(sale.totalVenta || 0)}</strong></div>
        <div class="ticket-line"><span>Medio declarado</span><strong>${esc(methods || "—")}</strong></div>
        ${hasProducts ? '<div class="ticket-line"><span>Stock</span><strong>No consultado ni modificado</strong></div>' : ''}
      `;
    } else if (!state.items.length) {
      container.innerHTML = '<p style="color:var(--muted);">Añade conceptos del catálogo.</p>';
    } else {
      container.innerHTML = state.items.map((item, i) => {
        const lineTotal = quoteLines[i]?.totalLinea;
        const displayName = quoteLines[i]?.concepto || item.name;
        const displayTotal = lineTotal !== undefined
          ? money(lineTotal)
          : (state.quoteLoading ? "Calculando…" : "No disponible");
        return `
          <div class="ticket-line">
            <span>${esc(displayName)}</span>
            <span class="qty">
              <button data-action="minus" data-idx="${i}">−</button>
              <span>${item.qty}</span>
              <button data-action="plus" data-idx="${i}">+</button>
            </span>
            <span>${displayTotal}</span>
          </div>
        `;
      }).join("");
      container.querySelectorAll("button[data-action]").forEach((btn) => {
        btn.onclick = () => {
          const idx = Number(btn.dataset.idx);
          if (btn.dataset.action === "plus") state.items[idx].qty++;
          else if (btn.dataset.action === "minus") {
            state.items[idx].qty--;
            if (state.items[idx].qty <= 0) state.items.splice(idx, 1);
          }
          state.payments = [];
          renderTicket();
          refreshSaleQuote();
        };
      });
    }

    const quote = state.saleQuote;
    const hasQuote = quote && Number.isFinite(Number(quote.totalVenta));
    $("subtotal").textContent = hasQuote ? money(quote.subtotalNeto) : (state.items.length ? (state.quoteLoading ? "Calculando…" : "—") : money(0));
    $("tax").value = hasQuote ? money(quote.impuestosTotal) : (state.items.length ? (state.quoteLoading ? "Calculando…" : "—") : money(0));
    $("total").textContent = hasQuote ? money(quote.totalVenta) : (state.items.length ? (state.quoteLoading ? "Calculando…" : "—") : money(0));

    const readiness = $("saleReadiness");
    if (hasQuote) {
      const blockers = quote.saleReadinessBlockers || [];
      const messages = [];
            if (blockers.includes("WIX_TAX_REGION_EXCLUDES_TAX")) messages.push("Wix trata los precios del catálogo como importes sin IVA; se calcula el impuesto por separado.");
      if (blockers.includes("PAYMENT_NOT_VERIFIED")) messages.push("Pago no verificado.");
      if (blockers.includes("TPV_WRITE_V1_TAX_RATE_UNVERIFIABLE")) messages.push("La tasa impositiva de alguna línea no puede verificarse.");
      if (blockers.includes("TPV_WRITE_V1_MULTI_VAT_UNSUPPORTED")) messages.push("La venta combina varias tasas de IVA; la escritura V1 solo admite una.");
      if (blockers.includes("TPV_WRITE_V1_SPLIT_TENDER_UNSUPPORTED")) messages.push("La escritura V1 admite un único medio de pago por venta.");
      if (blockers.includes("TPV_WRITE_V1_PAYMENT_METHOD_UNSUPPORTED")) messages.push("El medio de pago no está habilitado en el contrato V1.");
      if (blockers.includes("TPV_WRITE_V1_PAYMENT_TOTAL_MISMATCH")) messages.push("El importe aplicado del pago no coincide con el total de la venta.");
      if (blockers.includes("INVENTORY_POLICY_NOT_CONFIGURED")) messages.push("Inventario de productos no configurado en la ubicación física del salón.");
      if (blockers.includes("CONSUMER_PRICE_PRESENTATION_UNVERIFIED")) messages.push("Presentación del precio final al consumidor pendiente de validación.");
      readiness.textContent = quote.commercialRecordReady === true && state.operationalWritesAvailable === true
        ? "Registro comercial disponible; se guardará pendiente de cobro. " + messages.join(" ")
        : "No se puede registrar la venta: " + (messages.join(" ") || "faltan requisitos de catálogo o sesión.");
    } else if (state.quoteLoading) {
      readiness.textContent = "Consultando catálogo y cálculo fiscal de Wix…";
    } else if (state.quoteError) {
      readiness.textContent = "No se pudo preparar la venta: " + publicErrorMessage(state.quoteError);
    } else {
      readiness.textContent = "Añade líneas para calcular el total fiscal.";
    }
    renderPaymentSummary();
  }

  $("clearTicket").onclick = () => {
    state.quoteSequence++;
    state.items = [];
    state.payments = [];
    state.saleQuote = null;
    state.quoteLoading = false;
    state.quoteError = null;
    renderTicket();
  };

  /* ---------- Pago: controles condicionados a capability backend ---------- */
  function paymentTotals() {
    const totalsMinor = Object.fromEntries(PAYMENT_METHODS.map((method) => [method, 0]));
    state.payments.forEach((payment) => {
      totalsMinor[payment.method] += Math.max(0, payment.amountMinor - payment.changeMinor);
    });
    return Object.fromEntries(
      PAYMENT_METHODS.map((method) => [method, totalsMinor[method] / 100])
    );
  }

  function renderPaymentSummary() {
    const totalMinor = state.saleQuote && Number.isFinite(Number(state.saleQuote.totalVenta))
      ? Math.round(Number(state.saleQuote.totalVenta) * 100)
      : null;
    const appliedMinor = state.payments.reduce((sum, payment) =>
      sum + payment.amountMinor - payment.changeMinor, 0);
    const pendingMinor = totalMinor === null ? null : Math.max(0, totalMinor - appliedMinor);
    const changeMinor = state.payments.reduce((sum, payment) => sum + payment.changeMinor, 0);
    const paymentReady = state.saleQuote?.catalogVerified === true && state.saleQuote?.taxVerified === true && state.saleQuote?.commercialRecordReady === true && state.operationalWritesAvailable === true && state.capabilities.acceptPayment === true;
    $("paymentAmount").disabled = !paymentReady;
    $("addPayment").disabled = !paymentReady;
    document.querySelectorAll("[data-cash]").forEach((button) => { button.disabled = !paymentReady; });
    $("paymentMethods").querySelectorAll("button[data-method]").forEach((button) => {
      const unsupportedTender = button.dataset.method !== "EFECTIVO";
      button.disabled = !paymentReady || unsupportedTender;
      if (unsupportedTender && !button.title) button.title = "Medio de pago pendiente de integración.";
    });
    $("applied").textContent = money(appliedMinor / 100);
    $("pending").textContent = pendingMinor === null ? (state.quoteLoading ? "Calculando…" : "—") : money(pendingMinor / 100);
    $("change").textContent = money(changeMinor / 100);

    const totals = paymentTotals();
    const rows = state.payments.map((payment, index) => `
      <div class="payment-row" style="display:flex;align-items:center;gap:8px;padding:6px 0;">
        <span style="flex:1;">${esc(payment.method)}</span>
        <strong>${money(payment.amountMinor / 100)}</strong>
        <small>Aplicado: ${money((payment.amountMinor - payment.changeMinor) / 100)}${payment.changeMinor ? " · Cambio: " + money(payment.changeMinor / 100) : ""}</small>
        <button type="button" class="btn danger" data-remove-payment="${index}">Quitar</button>
      </div>
    `).join("");
    const breakdown = Object.entries(totals)
      .filter(([, amount]) => amount > 0)
      .map(([method, amount]) => `<div>${esc(method)}: ${money(amount)}</div>`)
      .join("");
    $("paymentList").innerHTML = rows
      ? rows + (breakdown ? `<hr><strong>Desglose aplicado:</strong>${breakdown}` : "")
      : "Sin pagos añadidos.";
    $("paymentList").querySelectorAll("[data-remove-payment]").forEach((button) => {
      button.onclick = () => {
        state.payments.splice(Number(button.dataset.removePayment), 1);
        refreshSaleQuote();
      };
    });
    const saleButton = $("saveSale");
    if (saleButton) {
      const cashConfirmed = state.payments.some((payment) =>
        payment.method === "EFECTIVO" && payment.amountMinor - payment.changeMinor > 0
      );
      saleButton.textContent = cashConfirmed
        ? "Confirmar cobro y registrar venta"
        : "Registrar pedido pendiente";
      saleButton.title = cashConfirmed
        ? "Al continuar confirmas que has recibido físicamente el efectivo."
        : "El pedido se registrará en Wix sin marcarlo como cobrado.";
    }
  }

  $("paymentMethods").querySelectorAll("button[data-method]").forEach((btn) => {
    btn.onclick = () => {
      $("paymentMethods").querySelectorAll("button").forEach((b) => b.classList.remove("active"));
      btn.classList.add("active");
      state.paymentMethod = btn.dataset.method;
    };
  });

  $("quickCash").querySelectorAll("button[data-cash]").forEach((btn) => {
    btn.onclick = () => {
      const val = btn.dataset.cash;
      if (val === "exact") {
        if (!state.saleQuote || state.saleQuote.catalogVerified !== true || state.saleQuote.taxVerified !== true || state.saleQuote.commercialRecordReady !== true || state.operationalWritesAvailable !== true || state.capabilities.acceptPayment !== true) {
          toast("No hay una cotización fiscal válida para la venta.");
          return;
        }
        const total = Number(state.saleQuote.totalVenta);
        $("paymentAmount").value = total.toFixed(2);
      } else {
        $("paymentAmount").value = val;
      }
    };
  });

  $("addPayment").onclick = () => {
    if (!state.saleQuote || state.saleQuote.catalogVerified !== true || state.saleQuote.taxVerified !== true || state.saleQuote.commercialRecordReady !== true || state.operationalWritesAvailable !== true) {
      toast("No se aceptan pagos: el backend no ha habilitado la capacidad de cobro.");
      return;
    }
    if (state.payments.length >= 1) {
      toast("El contrato V1 admite un único medio de pago. Quita el anterior para sustituirlo.");
      return;
    }
    const amount = num($("paymentAmount").value);
    if (!Number.isFinite(amount) || amount <= 0 || Math.abs(amount * 100 - Math.round(amount * 100)) > 1e-7) {
      toast("Introduce un importe válido con un máximo de dos decimales.");
      return;
    }
    if (!PAYMENT_METHODS.includes(state.paymentMethod)) {
      toast("Selecciona un medio de pago válido.");
      return;
    }
    const amountMinor = Math.round(amount * 100);
    const totalMinor = Math.round(Number(state.saleQuote.totalVenta) * 100);
    const appliedMinor = state.payments.reduce((sum, payment) =>
      sum + payment.amountMinor - payment.changeMinor, 0);
    const remainingMinor = totalMinor - appliedMinor;
    if (!state.items.length || remainingMinor <= 0) {
      toast("No hay importe pendiente para asignar.");
      return;
    }
    if (state.paymentMethod !== "EFECTIVO" && amountMinor > remainingMinor) {
      toast("Este medio de pago no admite cambio ni puede superar el pendiente.");
      return;
    }
    const changeMinor = state.paymentMethod === "EFECTIVO"
      ? Math.max(0, amountMinor - remainingMinor)
      : 0;
    state.payments.push({
      method: state.paymentMethod,
      amountMinor,
      changeMinor,
    });
    $("paymentAmount").value = "";
    renderPaymentSummary();
    refreshSaleQuote();
    if (changeMinor > 0) toast("Importe añadido. Cambio calculado: " + money(changeMinor / 100));
  };

  /* ---------- Registrar pedido y, si se ha confirmado efectivo, registrar el cobro nativo ---------- */
  $("saveSale").onclick = () => runOperation(async () => {
    if (!state.items.length) {
      toast("El ticket está vacío.");
      return;
    }
    if (!state.saleQuote || state.saleQuote.commercialRecordReady !== true || state.saleQuote.saleReady !== true || state.operationalWritesAvailable !== true) {
      toast("No se puede registrar la venta: faltan catálogo, cálculo fiscal o permiso de escritura.");
      return;
    }
    const result = await keyedAction("SALE", "SALE_REGISTER", {
      sessionId: state.session?.sessionId,
      lineItems: state.items.map((item) => ({
        tipoLinea: item.tipoLinea,
        itemId: item.id,
        varianteId: item.varianteId,
        cantidad: item.qty,
        profesionalId: state.profesionalId,
      })),
      medioPago: state.paymentMethod || null,
      contactId: state.customerIdentityId || null,
      payments: state.payments.map((payment) => ({
        method: payment.method,
        amount: payment.amountMinor / 100,
        change: payment.changeMinor / 100,
      })),
    }, "SALE_REGISTER_RES");
    if (result?.persisted !== true) throw new Error("TPV_SALE_NOT_PERSISTED");
    state.lastSale = result;
    if (result.paymentStatus === "PAID" && result.orderPersisted === true &&
        result.paymentPersisted === true && result.ledgerPersisted === true &&
        result.auditPersisted === true) {
      toast("Venta cobrada y registrada en Wix.");
    } else if (result.requiresReconciliation === true) {
      toast("Operación registrada, pero requiere conciliación. No repitas la venta; referencia: " +
        (result.idempotencyKey || "consulta TransaccionesTPV"));
    } else {
      toast("Pedido Wix registrado. El cobro queda pendiente.");
    }
    state.items = [];
    state.payments = [];
    state.saleQuote = null;
    renderTicket();
  }).catch((saleError) => error(saleError));

  /* ---------- Apertura de caja: sesión, movimiento, recibo y continuidad ---------- */
  function openingReceiptHtml(receipt) {
    const timeZone = receipt.timeZone || "Europe/Madrid";
    const openedAt = new Date(receipt.openedAt);
    if (Number.isNaN(openedAt.getTime())) throw new Error("TPV_OPENING_RECEIPT_DATE_INVALID");
    const dateLabel = receipt.operationDate || new Intl.DateTimeFormat("en-CA", {
      timeZone, year: "numeric", month: "2-digit", day: "2-digit"
    }).format(openedAt);
    const timeLabel = new Intl.DateTimeFormat("es-ES", {
      timeZone, hour: "2-digit", minute: "2-digit", second: "2-digit"
    }).format(openedAt);
    const breakdown = receipt.breakdown || {};
    const rows = ["EFECTIVO", "TARJETA", "BIZUM", "BONO", "TARJETA_REGALO", "OTROS"];
    return `
      <article class="ticket-receipt" id="openingTicketReceipt">
        <h3>Marian Madrid</h3>
        <p>APERTURA DE CAJA</p>
        <p>SESIÓN ABIERTA</p>
        <div class="ticket-divider"></div>
        <p>Profesional: <strong>${esc(receipt.professional || "—")}</strong></p>
        <p>Fecha: ${esc(dateLabel)}</p>
        <p>Hora: ${esc(timeLabel)}</p>
        <p>Sesión: ${esc(receipt.sessionId || "—")}</p>
        <div class="ticket-divider"></div>
        <div class="ticket-row"><span>TOTAL INICIAL</span><strong>${esc(money(receipt.total))}</strong></div>
        <div class="ticket-divider"></div>
        ${rows.map((method) => `<div class="ticket-row"><span>${esc(method.replace("_", " "))}</span><strong>${esc(money(breakdown[method] ?? 0))}</strong></div>`).join("")}
        <div class="ticket-divider"></div>
        <p>Movimiento: ${esc(receipt.movementId || "—")}</p>
        <p>transactionId: ${esc(receipt.transactionId || "—")}</p>
      </article>
      <button id="printOpeningTicket" type="button" class="btn" style="width:100%;margin-top:12px;">Imprimir ticket y activar cajón</button>
      <p class="status">El ticket se envía a impresión al confirmar la apertura. Para que se abra el cajón físico, el controlador Epson debe tener activada la opción de abrir el cajón al iniciar la impresión.</p>
    `;
  }

  $("openCash").onclick = async () => {
    if (state.busy) return;
    setBusy(true);
    try {
      const current = await call("CASHIER_STATE", {}, "CASHIER_STATE_RES");
      renderSession(current);
      const preview = current?.nextSessionPreview;
      if (current?.canOpenSession !== true || !preview?.sessionId || !preview?.operationDate) {
        throw new Error("TPV_SESSION_OPEN_NOT_ALLOWED");
      }
      const authenticatedOperatorId = current.authenticatedOperatorId;
      const authenticatedName = current.authenticatedDisplayName || authenticatedOperatorId || "Usuario autenticado";
      const preparedAt = preview.preparedAt || current.now;
      const preparedDate = preparedAt ? new Date(preparedAt) : null;
      const preparedTime = preparedDate && !Number.isNaN(preparedDate.getTime())
        ? new Intl.DateTimeFormat("es-ES", {
            timeZone: preview.timeZone || current.timeZone || "Europe/Madrid",
            hour: "2-digit", minute: "2-digit", second: "2-digit"
          }).format(preparedDate)
        : "—";
      const projectionLabel = current.cajaActual?.projectionStatus !== "LEGACY_UNVERIFIED"
        ? "La proyección actual se actualizará con la apertura."
        : "La proyección anterior se sustituirá; sus importes no se reutilizarán.";
      openModal("Abrir caja", `
        <p>Profesional / usuario de sesión: <strong>${esc(authenticatedName)}</strong></p>
        <p>Fecha de operación: <strong>${esc(preview.operationDate)}</strong></p>
        <p>Hora del sistema: <strong>${esc(preparedTime)}</strong></p>
        <p>Sesión: <strong>${esc(preview.sessionId)}</strong></p>
        <p>${esc(projectionLabel)}</p>
        <label class="status">Fondo inicial en efectivo
          <input id="openingCash" class="field" type="text" inputmode="decimal" placeholder="Importe (ej. 150,00)">
        </label>
        <label class="status">Motivo (opcional)
          <input id="openingReason" class="field" maxlength="200" placeholder="Apertura de caja">
        </label>
      `, async () => {
        const openingCashInput = $("openingCash").value;
        const openingCash = num(openingCashInput);
        if (!String(openingCashInput || "").trim() || !Number.isFinite(openingCash) || openingCash < 0) {
          throw new Error("TPV_INVALID_AMOUNT");
        }
        const expectedSessionId = preview.sessionId;
        if (state.busy) return { keepOpen: true };
        setBusy(true);

        let openedSessionId = null;
        let confirmed = null;
        try {
          const latest = await call("CASHIER_STATE", {}, "CASHIER_STATE_RES");
          renderSession(latest);
          if (latest?.canOpenSession !== true ||
              latest?.nextSessionPreview?.sessionId !== expectedSessionId ||
              latest?.authenticatedOperatorId !== authenticatedOperatorId) {
            throw new Error("TPV_SESSION_PREVIEW_STALE");
          }

          const opened = await keyedAction("SESSION_OPEN", "SESSION_OPEN", {
            expectedSessionId,
            openingCash,
            reason: $("openingReason").value.trim(),
            terminalId: "TPV",
          }, "SESSION_OPEN_RES");
          openedSessionId = opened?.sessionId;
          if (!openedSessionId) throw new Error("TPV_SESSION_OPEN_NOT_CONFIRMED");

          confirmed = await refreshCashState();
          if (confirmed?.operationalStatus !== "ABIERTA" ||
              confirmed?.sessionId !== openedSessionId ||
              confirmed?.authenticatedOperatorId !== authenticatedOperatorId ||
              Math.round(Number(confirmed?.expectedCash || 0) * 100) !== Math.round(openingCash * 100)) {
            throw new Error("TPV_SESSION_OPEN_NOT_CONFIRMED");
          }

          const receipt = opened.openingReceipt;
          if (!receipt || receipt.ticketType !== "APERTURA_CAJA" ||
              receipt.sessionId !== openedSessionId ||
              !receipt.movementId || !receipt.transactionId ||
              Math.round(Number(receipt.total) * 100) !== Math.round(openingCash * 100)) {
            throw new Error("TPV_OPENING_RECEIPT_INCOMPLETE");
          }

          $("modalTitle").textContent = "Apertura confirmada · Ticket de apertura";
          $("modalBody").innerHTML = openingReceiptHtml(receipt);
          $("modalConfirm").textContent = "Cerrar y continuar";
          $("modalCancel").classList.add("hidden");
          $("modalConfirm").onclick = closeModal;
          $("printOpeningTicket").onclick = () => window.print();
          if (opened.replayed !== true) {
            window.setTimeout(() => window.print(), 150);
          }
          toast("Caja abierta, movimiento registrado y ticket de apertura generado.");
          return { keepOpen: true };
        } catch (operationError) {
          if (openedSessionId) {
            $("modalTitle").textContent = "Apertura realizada · Verificación pendiente";
            $("modalBody").innerHTML = `
              <p>La apertura fue aceptada para la sesión <strong>${esc(openedSessionId)}</strong>.</p>
              <p>No se ha podido confirmar toda la proyección o el ticket: ${esc(publicErrorMessage(operationError))}</p>
              <p>No repitas la apertura con una clave nueva. Comprueba MovimientosCaja y CajaActual antes de continuar.</p>
            `;
            $("modalConfirm").textContent = "Cerrar";
            $("modalCancel").classList.add("hidden");
            $("modalConfirm").onclick = closeModal;
            toast("Apertura realizada; conciliación de la anotación pendiente.");
            return { keepOpen: true };
          }
          throw operationError;
        } finally {
          setBusy(false);
        }
      });
    } catch (e) {
      error(e);
    } finally {
      setBusy(false);
    }
  };

  /* ---------- Entrada / Salida de efectivo ---------- */
  function openCashMovement(type) {
    openModal(
      type === "ENTRADA" ? "Entrada de efectivo" : "Salida de efectivo",
      `
        <input id="movementAmount" class="field" type="text" inputmode="decimal" placeholder="Importe (ej. 25,50)">
        <input id="movementConcept" class="field" maxlength="500" placeholder="Concepto obligatorio">
      `,
      async () => {
        const amount = num($("movementAmount").value);
        const reason = $("movementConcept").value.trim();
        if (!Number.isFinite(amount) || amount <= 0 || reason.length < 3) {
          throw new Error("Indica importe y concepto válidos.");
        }
        await keyedAction("CASH_" + type, "CASH_MOVEMENT", {

  function csvCell(value) {
    let text = value === null || value === undefined
      ? ""
      : (typeof value === "object" ? JSON.stringify(value) : String(value));
    if (/^[=+\-@\t\r]/.test(text)) text = "'" + text;
    return "\"" + text.replaceAll("\"", "\"\"") + "\"";
  }

  function downloadCsv(rows, filename, columns) {
    const csv = [
      columns.map(csvCell).join(";"),
      ...rows.map((row) => columns.map((column) => csvCell(row[column])).join(";")),
    ].join("\n");
    const blob = new Blob(["\ufeff" + csv], { type: "text/csv;charset=utf-8" });
    const url = URL.createObjectURL(blob);
    const link = document.createElement("a");
    link.href = url;
    link.download = filename;
    link.style.display = "none";
    document.body.appendChild(link);
    link.click();
    link.remove();
    setTimeout(() => URL.revokeObjectURL(url), 1000);
  }

  async function exportData(operationType, filename, columns) {
    const sessionId = state.session?.sessionId;
    if (!sessionId) throw new Error("No hay una sesión de caja seleccionada.");
    const result = await call(operationType, { sessionId }, operationType + "_RES");
    if (result?.truncated === true) throw new Error("La exportación está incompleta; no se ha generado el CSV.");
    const rows = Array.isArray(result?.items) ? result.items : [];
    if (!rows.length) {
      toast("No hay datos para exportar.");
      return;
    }
    downloadCsv(rows, filename, columns);
    toast("Exportación CSV generada: " + rows.length + " registros.");
  }

  /* ---------- Handshake y contexto del controlador Wix ---------- */
  const BRIDGE_PROTOCOL_VERSION = 1;
  const HANDSHAKE_RETRY_MS = 500;
  // Retry for 60 s; keep a 5 s margin before the 65 s Velo watchdog.
  const HANDSHAKE_MAX_ATTEMPTS = 120;
  let handshakeAttempts = 0;
  let handshakeTimer = null;
  let widgetInitialized = false;

  function initializeWidgetAfterHandshake() {
    if (widgetInitialized) return;
    widgetInitialized = true;
    if (handshakeTimer) {
      clearTimeout(handshakeTimer);
      handshakeTimer = null;
    }

    loadCatalog("SERVICES").catch((e) => error(e));
    call("CATALOG_STAFF", {}, "CATALOG_STAFF_RES").then((data) => {
      state.staff = data.items || [];
      renderProfessionals();
    }).catch((e) => error(e));
    refreshCashState().catch((e) => error(e));
  }

  function sendHandshake() {
    if (widgetInitialized) return;
    if (handshakeAttempts >= HANDSHAKE_MAX_ATTEMPTS) {
      handshakeTimer = null;
      $("connStatus").className = "dot error";
      $("connLabel").textContent = "Sin conexión con Wix";
      error("No se recibió respuesta del controlador TPV.");
      return;
    }

    handshakeAttempts += 1;
    try {
      postToParent({
        type: "MM_READY",
        version: BRIDGE_PROTOCOL_VERSION,
        nonce: HANDSHAKE_NONCE,
        messageId: "tpv-ready-" + crypto.randomUUID().replace(/-/g, ""),
      });
    } catch (originError) {
      handshakeTimer = null;
      $("connStatus").className = "dot error";
      $("connLabel").textContent = "Origen no autorizado";
      error(originError);
      return;
    }
    handshakeTimer = setTimeout(() => {
      handshakeTimer = null;
      sendHandshake();
    }, HANDSHAKE_RETRY_MS);
  }

  window.addEventListener("message", (event) => {
    if (!PARENT_ORIGIN) return;
    if (event.source !== window.parent || event.origin !== PARENT_ORIGIN) return;
    const data = event.data || {};
    if (data.type !== "MM_CONTEXT") return;

    if (handshakeTimer) {
      clearTimeout(handshakeTimer);
      handshakeTimer = null;
    }

    const context = data.payload || {};
    // Nonce obligatorio: el controlador DEBE ecohar el nonce enviado en MM_READY
    // (aceptamos data.nonce o payload.nonce por compatibilidad).
    const receivedNonce = data.nonce || context.nonce;
    if (!receivedNonce || receivedNonce !== HANDSHAKE_NONCE) {
      $("connStatus").className = "dot error";
      $("connLabel").textContent = "Nonce de handshake inválido";
      error({ code: "TPV_HANDSHAKE_NONCE_INVALID" });
      return;
    }

    const connected = context.status === "SUCCESS" &&
      context.data?.component === "htmlTPV" &&
      context.data?.protocolVersion === BRIDGE_PROTOCOL_VERSION;
    state.capabilities = context.data?.capabilities &&
      typeof context.data.capabilities === "object"
      ? context.data.capabilities
      : {};
    state.operationalWritesAvailable = context.data?.operationalWritesAvailable === true;
    $("connStatus").className = "dot " + (connected ? "ok" : "error");
    $("connLabel").textContent = connected ? (state.operationalWritesAvailable ? "Conectado · Operaciones autorizadas" : "Conectado · Solo lectura") : "No autorizado";

    if (connected) {
      initializeWidgetAfterHandshake();
    } else {
      error(context.error?.code || "No se pudo validar el contexto del TPV.");
    }
  });

  /* ---------- Inicialización ---------- */
  function init() {
    $("connStatus").className = "dot";
    $("connLabel").textContent = "Conectando…";
    updateControlAvailability();
    sendHandshake();
  }

  init();
