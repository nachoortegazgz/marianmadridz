<script>
/* ============================================================
   TPV Marian Madrid — Interfaz mejorada
   Interfaz TPV: contrato de comunicación con el controlador Wix.
   ============================================================ */
(function () {
  "use strict";

  const $ = (id) => document.getElementById(id);
  const esc = (s) => String(s ?? "").replace(/[&<>"']/g, (c) => ({
    "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;"
  }[c]));
  const num = (v) => Number.parseFloat(v) || 0;
  const money = (n) => num(n).toFixed(2).replace(".", ",") + " €";
  const PAYMENT_METHODS = ["EFECTIVO", "TARJETA", "BIZUM", "BONO", "TARJETA_REGALO", "OTROS"];

  /* ---------- Estado local ---------- */
  const state = {
    session: null,
    // profesionalId is the native Bookings resourceId exposed by MapaStaff.
    profesionalId: null,
    staff: [],
    items: [],
    payments: [],
    paymentMethod: "EFECTIVO",
    busy: false,
    lastSale: null,
    // Compatibility transport key: this value is a native Wix Contacts _id.
    customerIdentityId: null,
    customerLabel: null,
  };

  /* ---------- Toast ---------- */
  let toastTimer;
  function toast(msg) {
    const el = $("toast");
    el.textContent = msg;
    el.classList.add("show");
    clearTimeout(toastTimer);
    toastTimer = setTimeout(() => el.classList.remove("show"), 3000);
  }
  function error(msg) { toast("Error: " + msg); }

  /* ---------- Modales ---------- */
  function openModal(title, bodyHtml, onConfirm) {
    $("modalTitle").textContent = title;
    $("modalBody").innerHTML = bodyHtml;
    $("modalBackdrop").classList.remove("hidden");
    $("modalConfirm").onclick = async () => {
      try {
        await onConfirm();
        closeModal();
      } catch (e) {
        error(e.message);
      }
    };
  }
  function closeModal() {
    $("modalBackdrop").classList.add("hidden");
  }
  $("modalClose").onclick = closeModal;
  $("modalCancel").onclick = closeModal;

  /* ---------- Envío al controlador (postMessage) ---------- */
  function call(type, payload, responseType) {
    return new Promise((resolve, reject) => {
      const id = "req-" + crypto.randomUUID().replace(/-/g, "");
      let timer = null;
      const cleanup = () => {
        window.removeEventListener("message", handler);
        if (timer) clearTimeout(timer);
      };
      const handler = (event) => {
        if (event.source !== window.parent) return;
        const data = event.data || {};
        if (data.type === responseType && (data.messageId === id || data.requestId === id)) {
          cleanup();
          const response = data.payload || {};
          if (data.error || response.status === "ERROR") {
            reject(new Error(response.error?.code || data.error || "TPV_OPERATION_FAILED"));
          } else {
            resolve(response.data !== undefined ? response.data : response);
          }
        }
      };
      window.addEventListener("message", handler);
      timer = setTimeout(() => {
        cleanup();
        reject(new Error("TPV_TIMEOUT"));
      }, 15000);
      window.parent.postMessage({ type, payload, messageId: id, version: 1 }, "*");
    });
  }

  /* ---------- Idempotencia local ---------- */
  function keyedAction(keyPrefix, type, payload, responseType) {
    const storageKey = "tpv_key_" + keyPrefix;
    let key = sessionStorage.getItem(storageKey);
    if (!key) {
      key = keyPrefix + "_" + crypto.randomUUID().replace(/-/g, "");
      sessionStorage.setItem(storageKey, key);
    }
    const fullPayload = Object.assign({}, payload, { idempotencyKey: key });
    return call(type, fullPayload, responseType).then((result) => {
      sessionStorage.removeItem(storageKey);
      return result;
    });
  }

  /* ---------- Profesional seleccionado; el operador real procede del backend ---------- */
  function renderProfessionals() {
    const select = $("professionalSelect");
    select.innerHTML = '<option value="">Sin asignar</option>' +
      state.staff.map((person) =>
        `<option value="${esc(person.profesionalId)}">${esc(person.nombre)}</option>`
      ).join("");
    if (state.profesionalId) select.value = state.profesionalId;
  }

  $("professionalSelect").onchange = (event) => {
    const person = state.staff.find((item) => item.profesionalId === event.target.value);
    state.profesionalId = person ? person.profesionalId : null;
    toast(person ? "Profesional: " + person.nombre : "Profesional sin asignar");
  };

  function renderSelectedCustomer() {
    $("selectClient").textContent = state.customerIdentityId
      ? "Cliente: " + state.customerLabel
      : "Cliente";
  }

  async function searchCustomers() {
    const search = $("customerSearch").value.trim();
    if (search.length < 2) throw new Error("Introduce al menos dos caracteres.");
    const result = await call("CUSTOMERS_SEARCH", { search }, "CUSTOMERS_SEARCH_RES");
    const items = Array.isArray(result?.items) ? result.items : [];
    const container = $("customerResults");
    container.innerHTML = items.length
      ? items.map((item, index) => `
          <button type="button" class="btn" data-customer-index="${index}" style="display:block;width:100%;text-align:left;margin:6px 0;">
            ${esc(item.displayName)}${item.email ? " · " + esc(item.email) : ""}
          </button>
        `).join("")
      : "<p>No se encontraron clientes.</p>";
    container.querySelectorAll("[data-customer-index]").forEach((button) => {
      button.onclick = () => {
        const selected = items[Number(button.dataset.customerIndex)];
        // identityId is retained in the widget protocol; it contains the Wix Contact _id.
        if (!selected?.identityId) return;
        state.customerIdentityId = selected.identityId;
        state.customerLabel = selected.displayName;
        renderSelectedCustomer();
        closeModal();
        toast("Cliente seleccionado.");
      };
    });
  }

  $("selectClient").onclick = () => {
    openModal("Seleccionar cliente", `
      <input id="customerSearch" class="field" maxlength="100" placeholder="Nombre o correo del cliente">
      <button id="searchCustomers" type="button" class="btn">Buscar</button>
      <button id="clearCustomer" type="button" class="btn danger">Quitar cliente</button>
      <div id="customerResults" style="max-height:300px;overflow-y:auto;margin-top:10px;"></div>
    `, async () => {
      if (!state.customerIdentityId) throw new Error("Selecciona un cliente o utiliza Quitar cliente.");
    });
    $("searchCustomers").onclick = () => searchCustomers().catch((e) => error(e.message));
    $("customerSearch").onkeydown = (event) => {
      if (event.key === "Enter") searchCustomers().catch((e) => error(e.message));
    };
    $("clearCustomer").onclick = () => {
      state.customerIdentityId = null;
      state.customerLabel = null;
      renderSelectedCustomer();
      closeModal();
      toast("Cliente quitado.");
    };
  };

  function renderSession(session) {
    state.session = session || null;
    const operationalStatus = state.session?.operationalStatus || state.session?.status || "SIN_SESION";
    const isOpen = operationalStatus === "ABIERTA";
    const isClosed = operationalStatus === "CERRADA";
    $("sessionLabel").textContent = state.session?.sessionId || "—";
    const operatorId = state.session?.operatorId || state.session?.authenticatedOperatorId || "—";
    const operatorName = state.session?.operatorDisplayName || state.session?.authenticatedDisplayName || operatorId;
    $("operatorLabel").textContent = "Operador: " + operatorName + " · Usuario: " + operatorId;
    const timestamp = state.session?.now || state.session?.openedAt || state.session?.closedAt;
    const parsedTimestamp = timestamp ? new Date(timestamp) : null;
    const timeZone = state.session?.timeZone || "Europe/Madrid";
    $("cashTime").textContent = parsedTimestamp && !Number.isNaN(parsedTimestamp.getTime())
      ? "Hora: " + new Intl.DateTimeFormat("es-ES", { dateStyle: "short", timeStyle: "medium", timeZone }).format(parsedTimestamp)
      : "Hora: —";
    $("cashStatus").className = "dot " + (isOpen ? "ok" : "error");
    const integrationStatus = state.session?.integrationStatus || "NO_ENVIADO";
    const simulationStatus = state.session?.simulationStatus || (state.session?.isSynthetic ? "SYNTHETIC" : "NOT_APPLICABLE");
    $("cashLabel").textContent = isOpen
      ? "ABIERTA · Integración: " + integrationStatus + " · Simulación: " + simulationStatus
      : (isClosed ? "CERRADA · Integración: " + integrationStatus : "SIN SESIÓN");
    const canOpenSession = state.session?.canOpenSession ?? (operationalStatus === "SIN_SESION");
    $("openCash").disabled = isOpen || !canOpenSession || state.busy;
    ["saveSale", "cashIn", "cashOut", "refund", "partialCount", "ticketX", "ticketZ", "exportMovements", "exportAudit"].forEach((id) => {
      $(id).disabled = !isOpen || state.busy;
    });
  }

  async function refreshCashState() {
    const session = await call("CASHIER_STATE", {}, "CASHIER_STATE_RES");
    renderSession(session);
    return session;
  }

  /* ---------- Bloqueo de operaciones ---------- */
  function setBusy(value) {
    state.busy = value;
    const operationalStatus = state.session?.operationalStatus || state.session?.status || "SIN_SESION";
    const isOpen = operationalStatus === "ABIERTA";
    const canOpenSession = state.session?.canOpenSession ?? (operationalStatus === "SIN_SESION");
    $("openCash").disabled = value || isOpen || !canOpenSession;
    ["saveSale", "cashIn", "cashOut", "refund", "partialCount", "ticketX", "ticketZ", "exportMovements", "exportAudit"].forEach((id) => {
      const el = $(id);
      if (el) el.disabled = value || !isOpen;
    });
  }

  async function runOperation(fn) {
    if (state.busy) return;
    setBusy(true);
    try {
      const result = await fn();
      await refreshCashState();
      return result;
    } finally {
      setBusy(false);
    }
  }

  /* ---------- Catálogo nativo: Bookings y Stores Catalog V3 ---------- */
  function addCatalogLine(item, priceMinor, varianteId = null, variantLabel = "") {
    if (!Number.isSafeInteger(priceMinor) || priceMinor < 0) {
      toast("El artículo no tiene un precio válido.");
      return;
    }
    state.items.push({
      id: item.itemId,
      tipoLinea: item.tipoLinea || "PERSONALIZADA",
      name: [item.concepto, variantLabel].filter(Boolean).join(" · "),
      price: priceMinor / 100,
      qty: 1,
      varianteId,
      stockAfectado: false,
    });
    renderTicket();
  }

  async function selectCatalogItem(item) {
    if (item.tipoLinea === "PRODUCTO" && item.requiereSeleccionVariante === true) {
      const result = await call("CATALOG_VARIANTS", { productId: item.itemId }, "CATALOG_VARIANTS_RES");
      const variants = (result?.variantes || []).filter((variant) =>
        variant.disponible === true && Number.isSafeInteger(variant.precioMinor)
      );
      if (!variants.length) {
        toast("No hay variantes disponibles con precio válido.");
        return;
      }
      openModal("Seleccionar variante", `
        <select id="variantChoice" class="field">
          ${variants.map((variant) => `<option value="${esc(variant.varianteId)}">${esc(variant.concepto)} · ${money(variant.precioMinor / 100)}</option>`).join("")}
        </select>

  function renderCatalog(items) {
    const list = $("catalogList");
    if (!Array.isArray(items) || !items.length) {
      list.innerHTML = "<p>No hay elementos disponibles.</p>";

    }

        ? item.precioMinor

          : null);

  function renderTicket() {
    const container = $("ticketLines");
    if (!state.items.length) {
      container.innerHTML = '<p style="color:var(--muted);">Añade conceptos del catálogo.</p>';
    } else {
      container.innerHTML = state.items.map((item, i) => `
        <div class="ticket-line">
          <span>${esc(item.name)}</span>
          <span class="qty">
            <button data-action="minus" data-idx="${i}">−</button>
            <span>${item.qty}</span>
            <button data-action="plus" data-idx="${i}">+</button>
          </span>
          <span>${money(item.price * item.qty)}</span>
        </div>
      `).join("");
      container.querySelectorAll("button[data-action]").forEach((btn) => {
        btn.onclick = () => {
          const idx = Number(btn.dataset.idx);
          if (btn.dataset.action === "plus") state.items[idx].qty++;
          else if (btn.dataset.action === "minus") {
            state.items[idx].qty--;
            if (state.items[idx].qty <= 0) state.items.splice(idx, 1);
          }
          renderTicket();
        };
      });
    }
    const subtotal = state.items.reduce((s, it) => s + it.price * it.qty, 0);
    $("subtotal").textContent = money(subtotal);
    $("tax").value = money(0);
    $("total").textContent = money(subtotal);
    renderPaymentSummary();
  }

  /* ---------- Pagos ---------- */
  function paymentTotals() {
    const totalsMinor = Object.fromEntries(PAYMENT_METHODS.map((method) => [method, 0]));
    state.payments.forEach((payment) => {
      totalsMinor[payment.method] += Math.max(0, payment.amountMinor - payment.changeMinor);
    });
    return Object.fromEntries(
      PAYMENT_METHODS.map((method) => [method, totalsMinor[method] / 100])
    );
  }

  function renderPaymentSummary() {
    const totalMinor = Math.round(state.items.reduce((sum, item) => sum + item.price * item.qty, 0) * 100);
    const appliedMinor = state.payments.reduce((sum, payment) =>
      sum + payment.amountMinor - payment.changeMinor, 0);
    const pendingMinor = Math.max(0, totalMinor - appliedMinor);
    const changeMinor = state.payments.reduce((sum, payment) => sum + payment.changeMinor, 0);
    $("applied").textContent = money(appliedMinor / 100);
    $("pending").textContent = money(pendingMinor / 100);
    $("change").textContent = money(changeMinor / 100);

    const totals = paymentTotals();
    const rows = state.payments.map((payment, index) => `
      <div class="payment-row" style="display:flex;align-items:center;gap:8px;padding:6px 0;">
        <span style="flex:1;">${esc(payment.method)}</span>
        <strong>${money(payment.amountMinor / 100)}</strong>
        <small>Aplicado: ${money((payment.amountMinor - payment.changeMinor) / 100)}${payment.changeMinor ? " · Cambio: " + money(payment.changeMinor / 100) : ""}</small>
        <button type="button" class="btn danger" data-remove-payment="${index}">Quitar</button>
      </div>
    `).join("");
    const breakdown = Object.entries(totals)
      .filter(([, amount]) => amount > 0)
      .map(([method, amount]) => `<div>${esc(method)}: ${money(amount)}</div>`)
      .join("");
    $("paymentList").innerHTML = rows
      ? rows + (breakdown ? `<hr><strong>Desglose aplicado:</strong>${breakdown}` : "")
      : "Sin pagos añadidos.";
    $("paymentList").querySelectorAll("[data-remove-payment]").forEach((button) => {
      button.onclick = () => {
        state.payments.splice(Number(button.dataset.removePayment), 1);
        renderPaymentSummary();
      };
    });
  }

  $("addPayment").onclick = () => {
    const amount = Number($("paymentAmount").value);
    if (!Number.isFinite(amount) || amount <= 0 || Math.abs(amount * 100 - Math.round(amount * 100)) > 1e-7) {
      toast("Introduce un importe válido con un máximo de dos decimales.");
      return;
    }
    if (!PAYMENT_METHODS.includes(state.paymentMethod)) {
      toast("Selecciona un medio de pago válido.");
      return;
    }
    const amountMinor = Math.round(amount * 100);
    const totalMinor = Math.round(state.items.reduce((sum, item) => sum + item.price * item.qty, 0) * 100);
    const appliedMinor = state.payments.reduce((sum, payment) =>
      sum + payment.amountMinor - payment.changeMinor, 0);
    const remainingMinor = totalMinor - appliedMinor;
    if (!state.items.length || remainingMinor <= 0) {
      toast("No hay importe pendiente de cobro.");
      return;
    }
    if (state.paymentMethod !== "EFECTIVO" && amountMinor > remainingMinor) {
      toast("Este medio de pago no admite cambio ni puede superar el pendiente.");
      return;
    }
    const changeMinor = state.paymentMethod === "EFECTIVO"
      ? Math.max(0, amountMinor - remainingMinor)
      : 0;
    state.payments.push({
      method: state.paymentMethod,
      amountMinor,
      changeMinor,
    });
    $("paymentAmount").value = "";
    renderPaymentSummary();
    if (changeMinor > 0) toast("Pago añadido. Cambio: " + money(changeMinor / 100));
  };

  function csvCell(value) {
    let text = value === null || value === undefined
      ? ""
      : (typeof value === "object" ? JSON.stringify(value) : String(value));
    if (/^[=+\-@\t\r]/.test(text)) text = "'" + text;
    return "\"" + text.replaceAll("\"", "\"\"") + "\"";
  }

  function downloadCsv(rows, filename, columns) {
    const csv = [
      columns.map(csvCell).join(";"),
      ...rows.map((row) => columns.map((column) => csvCell(row[column])).join(";")),
    ].join("\n");
    const blob = new Blob(["\ufeff" + csv], { type: "text/csv;charset=utf-8" });
    const url = URL.createObjectURL(blob);
    const link = document.createElement("a");
    link.href = url;
    link.download = filename;
    link.style.display = "none";
    document.body.appendChild(link);
    link.click();
    link.remove();
    setTimeout(() => URL.revokeObjectURL(url), 1000);
  }

  async function exportData(operationType, filename, columns) {
    const sessionId = state.session?.sessionId;
    if (!sessionId) throw new Error("No hay una sesión de caja seleccionada.");
    const result = await call(operationType, { sessionId }, operationType + "_RES");
    if (result?.truncated === true) throw new Error("La exportación está incompleta; no se ha generado el CSV.");
    const rows = Array.isArray(result?.items) ? result.items : [];
    if (!rows.length) {
      toast("No hay datos para exportar.");
      return;
    }
    downloadCsv(rows, filename, columns);
    toast("Exportación CSV generada: " + rows.length + " registros.");
  }
