<script>
/* ============================================================
   TPV Marian Madrid — Interfaz mejorada
   Codificación neutra: sin referencias a sandbox/qa/live.
   ============================================================ */
(function () {
  "use strict";

  const $ = (id) => document.getElementById(id);
  const esc = (s) => String(s ?? "").replace(/[&<>"']/g, (c) => ({
    "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;"
  }[c]));
  const num = (v) => Number.parseFloat(v) || 0;
  const money = (n) => num(n).toFixed(2).replace(".", ",") + " €";

  /* ---------- Estado local ---------- */
  const state = {
    session: null,
    profesionalId: null,
    staff: [],
    items: [],
    payments: [],
    paymentMethod: "EFECTIVO",
    busy: false,
    lastSale: null,
  };

  /* ---------- Toast ---------- */
  let toastTimer;
  function toast(msg) {
    const el = $("toast");
    el.textContent = msg;
    el.classList.add("show");
    clearTimeout(toastTimer);
    toastTimer = setTimeout(() => el.classList.remove("show"), 3000);
  }
  function error(msg) { toast("Error: " + msg); }

  /* ---------- Modales ---------- */
  function openModal(title, bodyHtml, onConfirm) {
    $("modalTitle").textContent = title;
    $("modalBody").innerHTML = bodyHtml;
    $("modalBackdrop").classList.remove("hidden");
    $("modalConfirm").onclick = async () => {
      try {
        await onConfirm();
        closeModal();
      } catch (e) {
        error(e.message);
      }
    };
  }
  function closeModal() {
    $("modalBackdrop").classList.add("hidden");
  }
  $("modalClose").onclick = closeModal;
  $("modalCancel").onclick = closeModal;

  /* ---------- Envío al controlador (postMessage) ---------- */
  function call(type, payload, responseType) {
    return new Promise((resolve, reject) => {
      const id = "req-" + Math.random().toString(36).slice(2);
      const handler = (event) => {
        const data = event.data || {};
        if (data.type === responseType && (data.messageId === id || data.requestId === id)) {
          window.removeEventListener("message", handler);
          const response = data.payload || {};
          if (data.error || response.status === "ERROR") {
            reject(new Error(response.error?.code || data.error || "TPV_OPERATION_FAILED"));
          } else {
            resolve(response.data !== undefined ? response.data : response);
          }
        }
      };
      window.addEventListener("message", handler);
      window.parent.postMessage({ type, payload, messageId: id, version: 1 }, "*");
      setTimeout(() => {
        window.removeEventListener("message", handler);
        reject(new Error("TPV_TIMEOUT"));
      }, 15000);
    });
  }

  /* ---------- Idempotencia local ---------- */
  function keyedAction(keyPrefix, type, payload, responseType) {
    const storageKey = "tpv_key_" + keyPrefix;
    let key = sessionStorage.getItem(storageKey);
    if (!key) {
      key = keyPrefix + "_" + crypto.randomUUID().replace(/-/g, "");
      sessionStorage.setItem(storageKey, key);
    }
    const fullPayload = Object.assign({}, payload, { idempotencyKey: key });
    return call(type, fullPayload, responseType).then((result) => {
      sessionStorage.removeItem(storageKey);
      return result;
    });
  }

  /* ---------- Profesional seleccionado; el operador real procede del backend ---------- */
  function renderProfessionals() {
    const select = $("professionalSelect");
    select.innerHTML = '<option value="">Sin asignar</option>' +
      state.staff.map((person) =>
        `<option value="${esc(person.profesionalId)}">${esc(person.nombre)}</option>`
      ).join("");
    if (state.profesionalId) select.value = state.profesionalId;
  }

  $("professionalSelect").onchange = (event) => {
    const person = state.staff.find((item) => item.profesionalId === event.target.value);
    state.profesionalId = person ? person.profesionalId : null;
    toast(person ? "Profesional: " + person.nombre : "Profesional sin asignar");
  };

  function renderSession(session) {
    state.session = session || null;
    $("sessionLabel").textContent = state.session?.sessionId || "—";
    const isOpen = Boolean(state.session && state.session.status === "ABIERTA");
    $("cashStatus").className = "dot " + (isOpen ? "ok" : "error");
    $("cashLabel").textContent = isOpen
      ? "Caja abierta · operación sintética no conciliada"
      : "Caja cerrada";
    $("openCash").disabled = isOpen;
  }

  /* ---------- Bloqueo de operaciones ---------- */
  function setBusy(value) {
    state.busy = value;
    ["openCash", "saveSale", "cashIn", "cashOut", "refund", "ticketX", "ticketZ"].forEach((id) => {
      const el = $(id);
      if (el) el.disabled = value;
    });
  }

  async function runOperation(fn) {
    if (state.busy) return;
    setBusy(true);
    try {
      return await fn();
    } finally {
      setBusy(false);
    }
  }

  /* ---------- Catálogo nativo: Bookings y Stores Catalog V3 ---------- */
  function addCatalogLine(item, priceMinor, varianteId = null, variantLabel = "") {
    if (!Number.isSafeInteger(priceMinor) || priceMinor < 0) {
      toast("El artículo no tiene un precio válido.");
      return;
    }
    state.items.push({
      id: item.itemId,
      tipoLinea: item.tipoLinea || "PERSONALIZADA",
      name: [item.concepto, variantLabel].filter(Boolean).join(" · "),
      price: priceMinor / 100,
      qty: 1,
      varianteId,
      stockAfectado: false,
    });
    renderTicket();
  }

  async function selectCatalogItem(item) {
    if (item.tipoLinea === "PRODUCTO" && item.requiereSeleccionVariante === true) {
      const result = await call("CATALOG_VARIANTS", { productId: item.itemId }, "CATALOG_VARIANTS_RES");
      const variants = (result?.variantes || []).filter((variant) =>
        variant.disponible === true && Number.isSafeInteger(variant.precioMinor)
      );
      if (!variants.length) {
        toast("No hay variantes disponibles con precio válido.");
        return;
      }
      openModal("Seleccionar variante", `
        <select id="variantChoice" class="field">
          ${variants.map((variant) => `<option value="${esc(variant.varianteId)}">${esc(variant.concepto)} · ${money(variant.precioMinor / 100)}</option>`).join("")}
        </select>
      `, async () => {
        const selected = variants.find((variant) => variant.varianteId === $("variantChoice").value);
        if (!selected) throw new Error("TPV_VARIANT_REQUIRED");
        addCatalogLine(item, selected.precioMinor, selected.varianteId, selected.concepto);
      });
      return;
    }
    if (!Number.isSafeInteger(item.precioMinor)) {
      toast("El artículo no tiene un precio fijo configurado.");
      return;
    }
    addCatalogLine(item, item.precioMinor);
  }

  function renderCatalog(items) {
    const list = $("catalogList");
    if (!Array.isArray(items) || !items.length) {
      list.innerHTML = "<p>No hay elementos disponibles.</p>";
      return;
    }
    list.innerHTML = items.map((item, index) => {
      const priceMinor = Number.isSafeInteger(item.precioMinor)
        ? item.precioMinor
        : (item.precioFijo !== null && item.precioFijo !== undefined
          ? Math.round(Number(item.precioFijo) * 100)
          : null);
      const priceLabel = item.requiereSeleccionVariante === true
        ? "Seleccionar variante"
        : (Number.isSafeInteger(priceMinor) ? money(priceMinor / 100) : "Precio no configurado");
      return `
        <div class="catalog-item" data-index="${index}">
          <span>${esc(item.concepto || "Artículo")}</span>
          <span class="price">${esc(priceLabel)}</span>
        </div>
      `;
    }).join("");
    list.querySelectorAll(".catalog-item").forEach((element) => {
      element.onclick = () => {
        const item = items[Number(element.dataset.index)];
        selectCatalogItem(item).catch((error) => error.message === "TPV_TIMEOUT"
          ? toast("Tiempo de espera agotado.")
          : error(error.message));
      };
    });
  }

  async function loadCatalog(kind) {
    const type = kind === "PRODUCTS" ? "CATALOG_PRODUCTS" : "CATALOG_SERVICES";
    const responseType = type + "_RES";
    const result = await call(type, { limit: 100 }, responseType);
    renderCatalog(result?.items || []);
  }

  $("tabServicios").onclick = () => loadCatalog("SERVICES").catch((e) => error(e.message));
  $("tabProductos").onclick = () => loadCatalog("PRODUCTS").catch((e) => error(e.message));
  $("tabPersonalizada").onclick = () => openModal("Tarifa personalizada", `
    <input id="customConcept" class="field" maxlength="160" placeholder="Concepto">
    <input id="customPrice" class="field" type="number" min="0.01" step="0.01" placeholder="Precio">
  `, async () => {
    const concept = $("customConcept").value.trim();
    const price = num($("customPrice").value);
    if (concept.length < 2 || price <= 0) throw new Error("Indica concepto e importe válidos.");
    state.items.push({ id: null, tipoLinea: "PERSONALIZADA", name: concept, price, qty: 1, varianteId: null, stockAfectado: false });
    renderTicket();
  });
  $("refreshCatalog").onclick = () => loadCatalog("SERVICES").catch((e) => error(e.message));

  /* ---------- Ticket ---------- */
  function renderTicket() {
    const container = $("ticketLines");
    if (!state.items.length) {
      container.innerHTML = '<p style="color:var(--muted);">Añade conceptos del catálogo.</p>';
    } else {
      container.innerHTML = state.items.map((item, i) => `
        <div class="ticket-line">
          <span>${esc(item.name)}</span>
          <span class="qty">
            <button data-action="minus" data-idx="${i}">−</button>
            <span>${item.qty}</span>
            <button data-action="plus" data-idx="${i}">+</button>
          </span>
          <span>${money(item.price * item.qty)}</span>
        </div>
      `).join("");
      container.querySelectorAll("button[data-action]").forEach((btn) => {
        btn.onclick = () => {
          const idx = Number(btn.dataset.idx);
          if (btn.dataset.action === "plus") state.items[idx].qty++;
          else if (btn.dataset.action === "minus") {
            state.items[idx].qty--;
            if (state.items[idx].qty <= 0) state.items.splice(idx, 1);
          }
          renderTicket();
        };
      });
    }
    const subtotal = state.items.reduce((s, it) => s + it.price * it.qty, 0);
    $("subtotal").textContent = money(subtotal);
    $("taxes").textContent = money(0);
    $("total").textContent = money(subtotal);
    renderPaymentSummary();
  }

  $("clearTicket").onclick = () => {
    state.items = [];
    state.payments = [];
    renderTicket();
  };

  /* ---------- Pagos ---------- */
  function renderPaymentSummary() {
    const total = state.items.reduce((s, it) => s + it.price * it.qty, 0);
    const applied = state.payments.reduce((s, p) => s + p.amount, 0);
    const pending = Math.max(0, total - applied);
    const change = Math.max(0, applied - total);
    $("applied").textContent = money(applied);
    $("pending").textContent = money(pending);
    $("change").textContent = money(change);
    $("paymentList").innerHTML = state.payments.length
      ? state.payments.map((p) => `<div>${esc(p.method)}: ${money(p.amount)}</div>`).join("")
      : "Sin pagos añadidos.";
  }

  $("paymentMethods").querySelectorAll("button[data-method]").forEach((btn) => {
    btn.onclick = () => {
      $("paymentMethods").querySelectorAll("button").forEach((b) => b.classList.remove("active"));
      btn.classList.add("active");
      state.paymentMethod = btn.dataset.method;
    };
  });

  $("quickCash").querySelectorAll("button[data-cash]").forEach((btn) => {
    btn.onclick = () => {
      const val = btn.dataset.cash;
      if (val === "exact") {
        const total = state.items.reduce((s, it) => s + it.price * it.qty, 0);
        $("paymentAmount").value = total.toFixed(2);
      } else {
        $("paymentAmount").value = val;
      }
    };
  });

  $("addPayment").onclick = () => {
    const amount = num($("paymentAmount").value);
    if (amount <= 0) {
      toast("Introduce un importe válido.");
      return;
    }
    state.payments.push({ method: state.paymentMethod, amount });
    $("paymentAmount").value = "";
    renderPaymentSummary();
  };

  /* ---------- Guardar venta ---------- */
  $("saveSale").onclick = () => runOperation(async () => {
    
    if (!state.items.length) {
      toast("El ticket está vacío.");
      return;
    }
    const total = state.items.reduce((s, it) => s + it.price * it.qty, 0);
    const applied = state.payments.reduce((s, p) => s + p.amount, 0);
    if (Math.abs(applied - total) > 0.001) {
      toast("El pago no cubre el total exacto.");
      return;
    }
    try {
      const result = await keyedAction("SALE", "SALE_REGISTER", {
        sessionId: state.session?.sessionId,
        profesionalId: state.profesionalId,
        lines: state.items.map((it) => ({
          itemId: it.id,
          name: it.name,
          qty: it.qty,
          price: it.price,
        })),
        payments: state.payments.map((p) => ({
          paymentMethod: p.method,
          amount: p.amount,
        })),
      }, "SALE_REGISTER_RES");
      state.lastSale = result;
      toast("Venta registrada.");
      state.items = [];
      state.payments = [];
      renderTicket();
    } catch (e) {
      error(e.message);
    }
  });

  /* ---------- Apertura de caja ---------- */
  $("openCash").onclick = () => openModal("Abrir caja", `
    <input id="openingCash" class="field" type="number" min="0" step="0.01" placeholder="Fondo inicial">
    <input id="openingReason" class="field" maxlength="200" placeholder="Motivo (opcional)">
  `, async () => {
    const openingCash = num($("openingCash").value);
    if (openingCash < 0) throw new Error("El fondo inicial no puede ser negativo.");
    const session = await keyedAction("SESSION_OPEN", "SESSION_OPEN", {
      openingCash,
      reason: $("openingReason").value.trim(),
      terminalId: "TPV",
    }, "SESSION_OPEN_RES");
    renderSession(session);
    toast("Sesión sintética abierta.");
  });

  /* ---------- Entrada / Salida de efectivo ---------- */
  function openCashMovement(type) {
    
    openModal(
      type === "ENTRADA" ? "Entrada de efectivo" : "Salida de efectivo",
      `
        <input id="movementAmount" class="field" type="number" min="0.01" step="0.01" placeholder="Importe">
        <input id="movementConcept" class="field" maxlength="500" placeholder="Concepto obligatorio">
      `,
      async () => {
        const amount = num($("movementAmount").value);
        const reason = $("movementConcept").value.trim();
        if (amount <= 0 || reason.length < 3) {
          throw new Error("Indica importe y concepto válidos.");
        }
        await keyedAction("CASH_" + type, "CASH_MOVEMENT", {
          movementType: type,
          amount,
          reason,
          profesionalId: state.profesionalId,
        }, "CASH_MOVEMENT_RES");
        toast(type === "ENTRADA" ? "Entrada registrada." : "Salida registrada.");
      }
    );
  }
  $("cashIn").onclick = () => openCashMovement("ENTRADA");
  $("cashOut").onclick = () => openCashMovement("SALIDA");

  /* ---------- Devolución ---------- */
  $("refund").onclick = () => {
    
    openModal("Devolución sintética", `
      <input id="refundSaleId" class="field" placeholder="ID de venta">
      <input id="refundAmount" class="field" type="number" min="0.01" step="0.01" placeholder="Importe">
      <select id="refundMethod" class="field">
        <option value="EFECTIVO">Efectivo</option>
        <option value="TARJETA">Tarjeta</option>
        <option value="BIZUM">Bizum</option>
        <option value="OTROS">Otros</option>
      </select>
      <input id="refundReason" class="field" maxlength="500" placeholder="Motivo obligatorio">
    `, async () => {
      const saleId = $("refundSaleId").value.trim();
      const amount = num($("refundAmount").value);
      const paymentMethod = $("refundMethod").value;
      const reason = $("refundReason").value.trim();
      if (!saleId || amount <= 0 || reason.length < 3) {
        throw new Error("Completa venta, importe y motivo.");
      }
      await keyedAction("REFUND", "REFUND", {
        saleId,
        reason,
        profesionalId: state.profesionalId,
        payments: [{ paymentMethod, amount }],
      }, "REFUND_RES");
      toast("Devolución registrada.");
    });
  };

  /* ---------- Resumen e historial ---------- */
  $("cashSummary").onclick = async () => {
    try {
      const result = await call("SESSION_SUMMARY", { sessionId: state.session?.sessionId }, "SESSION_SUMMARY_RES");
      openModal("Resumen de caja", `
        <pre style="white-space:pre-wrap;font-size:12px;">${esc(JSON.stringify(result, null, 2))}</pre>
      `, async () => {});
    } catch (e) { error(e.message); }
  };

  $("audit").onclick = async () => {
    try {
      const result = await call("AUDIT_LIST", { sessionId: state.session?.sessionId }, "AUDIT_LIST_RES");
      const items = result.items || [];
      openModal("Historial de operaciones", `
        <div style="max-height:400px;overflow-y:auto;">
          ${items.length ? items.map((i) => `
            <div style="padding:8px 0;border-bottom:1px solid var(--line);">
              <strong>${esc(i.operationType || "Operación")}</strong>
              <span style="float:right;">${money(i.amount || 0)}</span><br>
              <small>${esc(i.status || "Registrada")}</small>
              <small>${esc(i.reason || "")}</small>
            </div>
          `).join("") : "<p>No hay operaciones.</p>"}
        </div>
      `, async () => {});
    } catch (e) { error(e.message); }
  };

  /* ---------- Control X ---------- */
  $("ticketX").onclick = () => runOperation(async () => {
    try {
      const result = await call("TICKET_X", { sessionId: state.session?.sessionId }, "TICKET_X_RES");
      openModal("Control X", `
        <pre style="white-space:pre-wrap;font-size:12px;">${esc(JSON.stringify(result, null, 2))}</pre>
      `, async () => {});
    } catch (e) { error(e.message); }
  });

  /* ---------- Cierre Z ---------- */
  $("ticketZ").onclick = () => {
    
    openModal("Cierre Z diario", `
      <p>El cierre bloqueará la sesión actual.</p>
      <input id="countedCash" class="field" type="number" min="0" step="0.01" placeholder="Efectivo contado">
      <input id="differenceReason" class="field" maxlength="200" placeholder="Motivo si existe diferencia">
    `, async () => {
      const countedCash = num($("countedCash").value);
      const differenceReason = $("differenceReason").value.trim();
      await keyedAction("TICKET_Z", "TICKET_Z", {
        sessionId: state.session?.sessionId,
        countedCash,
        differenceReason,
        profesionalId: state.profesionalId,
      }, "TICKET_Z_RES");
      toast("Cierre Z registrado.");
    });
  };

  /* ---------- Impresión ---------- */
  $("printLast").onclick = () => {
    if (!state.lastSale) {
      toast("No hay último ticket.");
      return;
    }
    window.print();
  };

  /* ---------- Handshake y contexto del controlador Wix ---------- */
  window.addEventListener("message", (event) => {
    const data = event.data || {};
    if (data.type === "MM_CONTEXT") {
      const context = data.payload || {};
      const connected = context.status === "SUCCESS";
      $("connStatus").className = "dot " + (connected ? "ok" : "error");
      $("connLabel").textContent = connected ? "Conectado" : "No autorizado";
    }
  });

  /* ---------- Inicialización ---------- */
  function init() {
    window.parent.postMessage({
      type: "MM_READY",
      version: 1,
      messageId: "tpv-ready-" + crypto.randomUUID().replace(/-/g, ""),
    }, "*");
    loadCatalog("SERVICES").catch((e) => error(e.message));
    call("CATALOG_STAFF", {}, "CATALOG_STAFF_RES").then((data) => {
      state.staff = data.items || [];
      renderProfessionals();
    }).catch((e) => error(e.message));
    call("CASHIER_STATE", {}, "CASHIER_STATE_RES").then(renderSession).catch((e) => error(e.message));
  }

  init();
})();
</script>
</body>
</html>
