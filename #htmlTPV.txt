<script>
(function(){
 const $=id=>document.getElementById(id);
 const money=value=>new Intl.NumberFormat('es-ES',{style:'currency',currency:'EUR'}).format(Number(value)||0);
 const date=value=>value?new Intl.DateTimeFormat('es-ES',{dateStyle:'short',timeStyle:'short',timeZone:'Europe/Madrid'}).format(new Date(value)):'—';
 const text=value=>value===null||value===undefined||value===''?'—':String(value);
 function requestState(){ $('feedback').textContent='Consultando estado autorizado…'; window.parent.postMessage({type:'TPV_REQUEST_STATE',version:1},'*'); }
 function render(response){
  if(!response||response.status!=='SUCCESS'||!response.data){$('status').textContent='No disponible';$('feedback').textContent=response?.error?.message||'No se pudo consultar la caja.';return;}
  const d=response.data,p=d.sessionPayments||{},last=d.lastOperation||{};
  $('status').textContent=text(d.cashRegisterStatus);
  $('cash').textContent=money(d.netAmount);
  $('cashAccum').textContent=money(d.cashAmount);
  $('cardAccum').textContent=money(d.cardAmount);
  $('otherAccum').textContent=money(d.otherAmount);
  $('payCash').textContent=money(p.EFECTIVO?.amount)+' · '+text(p.EFECTIVO?.ticketCount)+' mov.';
  $('payCard').textContent=money(p.TARJETA?.amount)+' · '+text(p.TARJETA?.ticketCount)+' mov.';
  $('payBizum').textContent=money(p.BIZUM?.amount)+' · '+text(p.BIZUM?.ticketCount)+' mov.';
  $('sessionStatus').textContent=text(d.cashRegisterStatus);
  $('operations').textContent=text(d.totalOperations);
  $('openedAt').textContent=date(d.openedAt);
  $('lastAt').textContent=date(d.lastMovementAt);
  $('openedBy').textContent=text(d.openedByName);
  $('difference').textContent=d.difference===null||d.difference===undefined?'—':money(d.difference);
  $('movementId').textContent=text(last.movementId);
  $('description').textContent=text(last.description);
  $('movementType').textContent=[last.movementType,last.paymentMethod].filter(Boolean).join(' / ')||'—';
  $('amount').textContent=last.amount===undefined?'—':money(last.amount);
  $('actor').textContent=last.actor?.verificationStatus==='VERIFIED_BACKEND'?text(last.actor.displayName):'No verificado en origen';
  $('sequence').textContent=[last.sequenceNumber,last.traceId].filter(Boolean).join(' / ')||'—';
  $('feedback').textContent='Datos recibidos del WebMethod getCashierState.';
 }
 window.addEventListener('message',event=>{
  const message=event.data;
  if(!message||typeof message!=='object')return;
  if(message.type==='TPV_STATE')render(message.payload);
  if(message.type==='TPV_ERROR'){$('status').textContent='Error';$('feedback').textContent=text(message.message);}
 });
 $('refresh').addEventListener('click',requestState);
 window.addEventListener('load',requestState);
})();
</script>
