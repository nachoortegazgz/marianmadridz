async function closingZ(){
 if(!state.session)throw new Error("Abre una sesión antes de cerrar.");
 const countedCash=number($("closingCash").value);
 const expectedCash=number(state.session.expectedCash);
 const diff=number(countedCash-expectedCash);
 const reason=$("closingReason").value.trim();
 if(diff!==0&&reason.length<3)throw new Error("Indica un motivo (mín. 3 caracteres) para la diferencia de "+money(diff)+".");
 const nextStr=$("nextOpeningCash").value;
 if(nextStr===""||nextStr===null)throw new Error("Indica el efectivo para la próxima apertura.");
 const nextOpening=number(nextStr);
 if(!Number.isFinite(nextOpening)||nextOpening<0)throw new Error("El efectivo para próxima apertura no es válido.");
 const denoms=calculateDenominations(countedCash);
 const idempotencyKey=state.operationKeys.ticketZ||(state.operationKeys.ticketZ=makeKey());
 let result;
 try{
  result=unwrap(await request("QA_TICKET_Z",{sessionId:state.session.sessionId,countedCash,differenceReason:reason,nextOpeningCash:nextOpening,operatorId:state.operator?.profesionalId||null,idempotencyKey},"QA_TICKET_Z_RES"));
 }catch(e){delete state.operationKeys.ticketZ;throw e}
 delete state.operationKeys.ticketZ;
 toast("Cierre Z completado: "+result.reportId);
 printReceiptHtml("Cierre Z",buildClosingTicketHtml(result,countedCash,expectedCash,nextOpening,denoms,reason));
 $("closingCash").value="0";$("closingReason").value="";$("nextOpeningCash").value="";
 await refreshSession();
}

async function saveSale(){
 if(!state.session||!state.items.length)throw new Error("Se necesita sesión abierta y al menos una línea.");
 const t=totals(),paid=number(state.payments.reduce((s,p)=>s+paymentApplied(p),0));if(number(paid)!==t.total)throw new Error("El pago debe cubrir exactamente el total.");
 const lineas=state.items.map((x,i)=>({lineaNumero:i+1,tipoLinea:x.tipoLinea,itemId:x.itemId,concepto:x.concepto,cantidad:x.cantidad,precioUnitario:x.precioUnitario,descuentoImporte:x.descuentoImporte,subtotalNeto:number(x.cantidad*x.precioUnitario-x.descuentoImporte),impuestosImporte:0,totalLinea:number(x.cantidad*x.precioUnitario-x.descuentoImporte),profesionalId:state.operator?.profesionalId||null,stockAfectado:x.stockAfectado}));
 const sale={idempotencyKey:state.operationKeys.sale||(state.operationKeys.sale=makeKey()),sessionId:state.session.sessionId,lineas,pagos:state.payments,subtotalNeto:t.subtotal,impuestosTotal:0,totalVenta:t.total,customerContactId:state.customer?.contactId||"",operatorId:state.operator?.profesionalId||null};
 if(t.tax!==0){sale.impuestosTotal=t.tax;throw new Error("El impuesto manual requiere asignación por línea.")}
 unwrap(await request("QA_PAYMENT_VALIDATE",{total:t.total,payments:state.payments},"QA_PAYMENT_VALIDATE_RES"));
 const valid=unwrap(await request("VALIDATE_SALE_DRAFT",{sale},"VALIDATE_SALE_DRAFT_RES"));
 if(valid.importePendiente!==0)throw new Error("El borrador conserva importe pendiente.");
 const result=unwrap(await request("QA_SALE_REGISTER",sale,"QA_SALE_REGISTER_RES"));
 state.lastSale={saleId:result.saleId,total:result.total,breakdown:result.breakdown||{}};delete state.operationKeys.sale;
 const firstMethod=Object.keys(result.breakdown||{}).find(k=>result.breakdown[k]>0)||"EFECTIVO";
 $("refundReceiptId").value=result.saleId;
 $("refundReceiptDate").value=new Date().toISOString().slice(0,10);
 $("refundMethod").value=firstMethod;
 $("refundAmount").value=String(result.breakdown?.[firstMethod]||result.total);
 toast("Venta guardada: "+result.saleId);
 state.items=[];state.payments=[];renderTicket();await refreshSession();
}

function calculateDenominations(amount){const cashDenominations=[50000,20000,10000,5000,2000,1000,500,200,100,50,20,10,5,2,1];let remaining=Math.round(Number(amount)*100);return cashDenominations.map(value=>{const count=Math.floor(remaining/value);remaining-=count*value;return{value,count}}).filter(x=>x.count)}
function renderDenominations(){

 else{const sign=diff>0?"+":"";diffEl.textContent="Diferencia vs esperado ("+money(expected)+"): "+sign+money(diff);diffEl.className="tiny text-bad"}
}
