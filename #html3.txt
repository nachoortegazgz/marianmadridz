    /* ================================================================
     * MM_CONFIRMATION_CONTEXT — contrato con Velo ConfirmacionReserva
     *
     *   1) HTML envía MM_CONFIRMATION_READY con setInterval (1s) hasta
     *      recibir MM_CONFIRMATION_CONTEXT o agotar 30s (safety cap).
     *   2) Velo responde con MM_CONFIRMATION_CONTEXT (payload plano,
     *      o { error: "..." } si el Velo falla antes de construir payload).
     *   3) HTML renderiza campos, guarda window.bookingData, HABILITA
     *      los botones de descarga y DETIENE el handshake.
     *   4) Botones de navegación envían MM_NAV con target lógico
     *      ("MIS_RESERVAS" | "HOME"); el Velo traduce a URLS reales.
     * ================================================================ */

    var MM_CONFIRMATION_READY   = "MM_CONFIRMATION_READY";
    var MM_CONFIRMATION_CONTEXT = "MM_CONFIRMATION_CONTEXT";
    var MM_NAV                  = "MM_NAV";

    var HANDSHAKE_INTERVAL_MS = 1000;
    var HANDSHAKE_MAX_MS      = 30000;

    function sendToParent(type, payload) {
        try {
            if (window.parent && window.parent !== window) {
                window.parent.postMessage(
                    { version: "1.0", type: type, payload: payload || {} },
                    "*"
                );
            }
        } catch (e) {
            try { console.warn("[ConfirmacionReserva-HTML] postMessage failed", e); } catch (_) {}
        }
    }

    function safeText(v) {
        return v == null ? "" : String(v).trim();
    }

    function getQueryParam(name) {
        try {
            return new URLSearchParams(window.location.search).get(name);
        } catch (_) {
            return null;
        }
    }

    /* P2-1: escape de caracteres especiales para iCal (RFC 5545 §3.3.11).
       Aplica a SUMMARY y DESCRIPTION. LOCATION ya se construye con \\, explícito. */
    function escapeICalText(text) {
        return String(text || "")
            .replace(/\\/g, "\\\\")
            .replace(/;/g, "\\;")
            .replace(/,/g, "\\,")
            .replace(/\r?\n/g, "\\n");
    }

    /* ----------------------------------------------------------------
     * Render desde el payload plano del Velo
     * Acepta { error: "..." } para mostrar fallo si el Velo no pudo
     * construir el payload (P3-2).
     * ---------------------------------------------------------------- */
    function renderConfirmation(payload) {
        var p = payload && typeof payload === "object" ? payload : {};

        /* P3-2: el Velo puede enviar { error: "..." } si falla antes de
           construir el payload. Mostramos mensaje y salimos. */
        if (p.error) {
            var subtitleEl = document.getElementById("subtitleText");
            if (subtitleEl) subtitleEl.innerText = p.error;
            document.getElementById("txtServiceTitle").innerText = "Error";
            try { console.warn("[ConfirmacionReserva-HTML] Error del Velo:", p.error); } catch (_) {}
            return;
        }

        var nested = p.booking && typeof p.booking === "object" ? p.booking : {};

        var bookingIdF1 = safeText(p.bookingIdF1 || nested.bookingIdF1 || nested.bookingId || p.bookingId) || "---";
        var bookingIdF2 = safeText(p.bookingIdF2 || nested.bookingIdF2 || "");
        var serviceId = safeText(p.serviceId || nested.serviceId || "---");
        var tituloServicio = safeText(p.tituloServicio || p.serviceName || nested.tituloServicio || "---");

        var fecha = safeText(p.fecha || nested.fecha || "---");
        var hora = safeText(p.hora || nested.hora || "---");
        var horaF2 = safeText(p.horaF2 || nested.horaF2 || "");
        var duracion = safeText(p.duracion || nested.duracion || "");
        var extras = safeText(p.extras || nested.extras || p.extrasText || "Ninguno");
        var appointmentNote = safeText(p.appointmentNote || nested.appointmentNote || p.nota || "");

        var rawPrecio = Number(
            p.total != null ? p.total :
            nested.total != null ? nested.total :
            nested.totalAmount != null ? nested.totalAmount : 0
        ) || 0;

        var esDual = p.isCombined === true || nested.isCombined === true || Boolean(bookingIdF2);
        var servicio = safeText(p.serviceName || tituloServicio);
        var estilistaDisplay = safeText(
            nested.staffName || nested.resourceName || p.staffName || ""
        ) || "ASIGNACIÓN AUTOMÁTICA";

        document.getElementById("txtServiceTitle").innerText = tituloServicio;
        document.getElementById("modalFecha").innerText = fecha;
        document.getElementById("modalHora").innerText =
            hora + (horaF2 ? " (Fase 2: " + horaF2 + ")" : "");
        document.getElementById("modalDuracion").innerText = duracion ? duracion + " min" : "---";
        document.getElementById("modalExtras").innerText = extras || "Ninguno";

            commentsBox.style.display = "none";

    function parseHourMinute(hora) {
        var parts = String(hora || "00:00").split(":");
        var h = parseInt(parts[0], 10);
        var m = parseInt(parts[1], 10);
        if (!Number.isFinite(h)) h = 0;
        if (!Number.isFinite(m)) m = 0;
        return { h: h, m: m };
    }

    function descargarCalendario() {
        var d = window.bookingData;
        if (!d) {
            try { console.warn("[ConfirmacionReserva-HTML] bookingData no disponible aún"); } catch (_) {}
            return;
        }

        /* 1) Parsear fecha — acepta DD/MM/YYYY o YYYY-MM-DD */
        var rawDate = String(d.fecha || "");
        var year, month, day;
        if (rawDate.indexOf("/") >= 0) {
            var p1 = rawDate.split("/");
            if (p1.length < 3) return;
            day   = String(p1[0]).padStart(2, "0");
            month = String(p1[1]).padStart(2, "0");
            year  = String(p1[2]);
        } else if (rawDate.indexOf("-") >= 0) {
            var p2 = rawDate.split("-");
            if (p2.length < 3) return;
            year  = String(p2[0]);
            month = String(p2[1]).padStart(2, "0");
            day   = String(p2[2]).padStart(2, "0");
        } else {
            try { console.warn("[ConfirmacionReserva-HTML] fecha no parseable:", rawDate); } catch (_) {}
            return;
        }

        /* 2) Parsear hora (HH:MM o HH:MM:SS) — sin bug de medianoche */
        var start = parseHourMinute(d.hora);
        var durationMin = parseInt(String(d.duracion), 10);
        if (!Number.isFinite(durationMin) || durationMin <= 0) durationMin = 60;

        /* 3) Calcular fin con rollover de día */
        var startTotal = start.h * 60 + start.m;
        var endTotal = startTotal + durationMin;
        var dayRollover = Math.floor(endTotal / (24 * 60));
        var endMinutesInDay = endTotal % (24 * 60);
        var endH = Math.floor(endMinutesInDay / 60);
        var endM = endMinutesInDay % 60;

        var endDay = day, endMonth = month, endYear = year;
        if (dayRollover > 0) {
            var dt = new Date(Date.UTC(parseInt(year, 10), parseInt(month, 10) - 1, parseInt(day, 10) + dayRollover));
            endYear  = String(dt.getUTCFullYear());
            endMonth = String(dt.getUTCMonth() + 1).padStart(2, "0");
            endDay   = String(dt.getUTCDate()).padStart(2, "0");
        }

        var startStamp = year + month + day + "T" + fmtIcalTime(start.h, start.m);
        var endStamp   = endYear + endMonth + endDay + "T" + fmtIcalTime(endH, endM);

        /* 4) Construir líneas con saltos REALES (\n en JS) y dejar que
              escapeICalText convierta a escape iCal (\n literal). */
        var summaryLines = ["Marian Madrid — " + d.tituloServicio];

        var descriptionLines = [
            "Cita: " + d.servicio + " (" + d.estilistaDisplay + "). Servicio ID: " +
                d.tituloServicio + " (F1: " + d.id + ")" +
                (d.esDual ? " F2: " + (d.idF2 || "siguiente") : ""),
            "Complementos: " + d.extras
        ];
        if (d.duracion) descriptionLines.push("Duración: " + d.duracion + " min");
        if (d.appointmentNote) descriptionLines.push("Comentarios: " + d.appointmentNote);

        var summary = escapeICalText(summaryLines.join("\n"));
        var description = escapeICalText(descriptionLines.join("\n"));

        /* P2-2: LOCATION también pasa por escape (comas + potenciales
           caracteres especiales si cambia la dirección). */
        var location = escapeICalText("Calle Jorge Juan, Peluquería Marian Madrid, Madrid");

        var icsMsg =
            "BEGIN:VCALENDAR\n" +
            "VERSION:2.0\n" +
            "PRODID:-//Marian Madrid//Booking Calendar//ES\n" +
            "BEGIN:VEVENT\n" +
            "UID:" + d.id + "@marianmadrid.es\n" +
            "DTSTAMP:" + startStamp + "\n" +
            "DTSTART:" + startStamp + "\n" +
            "DTEND:" + endStamp + "\n" +
            "SUMMARY:" + summary + "\n" +
            "DESCRIPTION:" + description + "\n" +
            "LOCATION:" + location + "\n" +
            "END:VEVENT\n" +
            "END:VCALENDAR";

        var blob = new Blob([icsMsg], { type: "text/calendar;charset=utf-8" });
        var url = URL.createObjectURL(blob);
        var a = document.createElement("a");
        a.href = url;
        a.download = "MarianMadrid_" + d.tituloServicio + "_" + year + month + day + ".ics";
        document.body.appendChild(a);
        a.click();
        document.body.removeChild(a);
        URL.revokeObjectURL(url);
        try { console.info("[ConfirmacionReserva-HTML] Agenda .ics descargada"); } catch (_) {}
    }

    /* ----------------------------------------------------------------
     * Handshake P2-1: setInterval(1s) hasta recibir CONTEXT o agotar 30s
     * ---------------------------------------------------------------- */
    function sendReady(attempt) {
        try { console.info("[ConfirmacionReserva-HTML] READY attempt=" + attempt); } catch (_) {}
        sendToParent(MM_CONFIRMATION_READY, { attempt: attempt || 1 });
    }

    function startHandshake() {
        handshakeAttempt = 1;
        sendReady(handshakeAttempt);
        handshakeStopTime = Date.now() + HANDSHAKE_MAX_MS;
        handshakeTimer = setInterval(function () {
            if (Date.now() > handshakeStopTime) {
                stopHandshake();
                try { console.warn("[ConfirmacionReserva-HTML] handshake timeout 30s"); } catch (_) {}
                return;
            }
            handshakeAttempt += 1;
            sendReady(handshakeAttempt);
        }, HANDSHAKE_INTERVAL_MS);
    }

    function stopHandshake() {
        if (handshakeTimer) {
            clearInterval(handshakeTimer);
            handshakeTimer = null;
        }
    }

    /* ----------------------------------------------------------------
     * Listener de mensajes del Velo
     * ---------------------------------------------------------------- */
    window.addEventListener("message", function (event) {
        var msg = event && event.data;
        if (!msg || typeof msg !== "object") return;
        if (msg.type === MM_CONFIRMATION_CONTEXT) {
            /* P2-1: paramos el handshake en cuanto llega el payload
               (éxito o error), ya no hace falta seguir reintentando. */
            stopHandshake();
            renderConfirmation(msg.payload);
        }
    });

    /* ----------------------------------------------------------------
     * Init
     * ---------------------------------------------------------------- */
    function init() {
        try { console.info("[ConfirmacionReserva-HTML] init"); } catch (_) {}

        /* P3-3: el fallback de query params SOLO se activa con un flag
           explícito para evitar falsos positivos si el sitio añade
           ?tituloServicio=... por tracking/marketing. */
        if (getQueryParam("__debug_standalone") === "1") {
            renderConfirmation({
                bookingIdF1: getQueryParam("bookingIdF1"),
                bookingIdF2: getQueryParam("bookingIdF2"),
                serviceId: getQueryParam("serviceId"),
                tituloServicio: getQueryParam("tituloServicio"),
                fecha: getQueryParam("fecha"),
                hora: getQueryParam("hora"),
                horaF2: getQueryParam("horaF2"),
                duracion: getQueryParam("duracion"),
                extras: getQueryParam("extras"),
                total: getQueryParam("precio"),
                appointmentNote: getQueryParam("appointmentNote"),
                isCombined: getQueryParam("esDual") === "true"
            });
            return; /* No hace falta handshake en modo debug */
        }

        /* Listeners de botones */
        var btnTicket = document.getElementById("btnTicket");
        var btnICS = document.getElementById("btnICS");
        var btnMis = document.getElementById("btnMisReservas");
        var btnInicio = document.getElementById("btnInicio");

        if (btnTicket) btnTicket.addEventListener("click", descargarTicket);
        if (btnICS) btnICS.addEventListener("click", descargarCalendario);
        if (btnMis) btnMis.addEventListener("click", function () { navegarA("MIS_RESERVAS"); });
        if (btnInicio) btnInicio.addEventListener("click", function () { navegarA("HOME"); });

        startHandshake();
    }

    if (document.readyState === "loading") {
        document.addEventListener("DOMContentLoaded", init, { once: true });
    } else {
        init();
    }
