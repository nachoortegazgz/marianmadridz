  <script>
    (function () {
      'use strict';

      const MESSAGE_TYPES = {
        READY:'MM_READY', CONTEXT:'MM_CONTEXT', NAV:'MM_NAV',
        CASHIER_STATE:'CASHIER_STATE', CASHIER_STATE_RES:'CASHIER_STATE_RES',
        Z_CLOSING:'Z_CLOSING', Z_CLOSING_RES:'Z_CLOSING_RES',
        FISCAL_SUMMARY:'FISCAL_SUMMARY', FISCAL_SUMMARY_RES:'FISCAL_SUMMARY_RES',
        FISCAL_BOOK:'FISCAL_BOOK', FISCAL_BOOK_RES:'FISCAL_BOOK_RES',
        DOCUMENT_PREPARED:'DOCUMENT_PREPARED', DOCUMENT_PREPARED_RES:'DOCUMENT_PREPARED_RES',
        CREATE_SNAPSHOT:'CREATE_SNAPSHOT', CREATE_SNAPSHOT_RES:'CREATE_SNAPSHOT_RES',
        DOCUMENT_DOWNLOAD:'DOCUMENT_DOWNLOAD', DOCUMENT_DOWNLOAD_RES:'DOCUMENT_DOWNLOAD_RES',
      };

      const VALID_TYPES = new Set(Object.values(MESSAGE_TYPES));

      let PARENT_ORIGIN = null;
      let traceId = 'admin_widget_' + Date.now().toString(36);
      let contextData = null;
      let fallbackTimer = null;

      const $ = (id) => document.getElementById(id);
      const errorBanner=$('errorBanner'), infoBanner=$('infoBanner'), warnBanner=$('warnBanner');
      const userInfo=$('userInfo'), roleBadge=$('roleBadge');




      const cashSaldo=$('cashSaldo'), cashOps=$('cashOps'), cashDiaKey=$('cashDiaKey');
      const btnRefreshCash=$('btnRefreshCash'), btnZClosing=$('btnZClosing'), cashTicket=$('cashTicket');

      const fiscalYear=$('fiscalYear'), fiscalQuarter=$('fiscalQuarter'), fiscalResult=$('fiscalResult');
      const btnFiscalSummary=$('btnFiscalSummary'), btnFiscalBook=$('btnFiscalBook');

      const btnLoadPrepared=$('btnLoadPrepared'), btnCreateSnapshot=$('btnCreateSnapshot'), documentsBody=$('documentsBody');
      let activeSnapshotOperationId = null;
      let activeSnapshotPeriod = null;


      function send(type, payload={}) {
        try { window.parent.postMessage({ type, payload: { ...payload, traceId } }, PARENT_ORIGIN || '*'); } catch (_) {}
      }
      function err(m){errorBanner.textContent=m;errorBanner.style.display='block';infoBanner.style.display='none';warnBanner.style.display='none';}
      function info(m){infoBanner.textContent=m;infoBanner.style.display='block';errorBanner.style.display='none';warnBanner.style.display='none';}
      function warn(m){warnBanner.textContent=m;warnBanner.style.display='block';errorBanner.style.display='none';infoBanner.style.display='none';}
      function clearBanners(){errorBanner.style.display='none';infoBanner.style.display='none';warnBanner.style.display='none';}

      function setLoading(btn, loading) {
        if (!btn) return;
        if (loading) {
          if (!btn.dataset.original) btn.dataset.original = btn.textContent;
          btn.disabled = true; btn.textContent = 'Procesando…';
        } else {
          btn.disabled = false; btn.textContent = btn.dataset.original || btn.textContent;
        }
      }

      function createOperationUuid() {
        if (!window.crypto || typeof window.crypto.getRandomValues !== 'function') return null;
        const bytes = new Uint8Array(16);
        window.crypto.getRandomValues(bytes);
        bytes[6] = (bytes[6] & 0x0f) | 0x40;
        bytes[8] = (bytes[8] & 0x3f) | 0x80;
        const hex = Array.from(bytes, (value) => value.toString(16).padStart(2, '0')).join('');
        return hex.slice(0, 8) + '-' + hex.slice(8, 12) + '-' + hex.slice(12, 16) + '-' +
          hex.slice(16, 20) + '-' + hex.slice(20);
      }

      function downloadTextFile(filename, content, mimeType) {
        const blob = new Blob([String(content || '')], { type: mimeType || 'text/plain;charset=utf-8' });
        const url = URL.createObjectURL(blob);
        const anchor = document.createElement('a');
        anchor.href = url;
        anchor.download = filename;
        document.body.appendChild(anchor);
        anchor.click();
        anchor.remove();
        setTimeout(() => URL.revokeObjectURL(url), 1000);
      }

      function money(v) {
        const n = Number(v);
        if (!Number.isFinite(n)) return '--';
        return new Intl.NumberFormat('es-ES',{style:'currency',currency:contextData?.currencyCode||'EUR'}).format(n);
      }

        /* ── Caja ── */
      function renderCashier(state) {
        if (!state) { cashSaldo.textContent='--'; cashOps.textContent='--'; cashDiaKey.textContent='--'; return; }
        // DTO canónico de cajas.web.js:getCashierState.
        cashSaldo.textContent = money(state.netAmount);
        cashOps.textContent = Number.isFinite(Number(state.totalOperations)) ? String(state.totalOperations) : '--';
        cashDiaKey.textContent = state.operationDate || '--';
      }
      function renderTicket(t) {
        if (!t) { cashTicket.style.display='none'; cashTicket.replaceChildren(); return; }
        cashTicket.style.display = 'block';
        cashTicket.replaceChildren();
        const appendLine = (value, style) => {
          const line = document.createElement('div');
          if (style) line.style.cssText = style;
          line.textContent = String(value);
          cashTicket.appendChild(line);
        };
        appendLine(t.title || 'TICKET', 'text-align:center;font-weight:900;text-transform:uppercase');
        if (t.ticketNumber) appendLine('Nº: ' + t.ticketNumber);
        if (t.concept) appendLine('Concepto: ' + t.concept);
        if (t.amount != null) appendLine('Importe: ' + money(t.amount));
        if (t.traceId) appendLine('Traza: ' + t.traceId);
        if (t.fiscalFingerprint) appendLine('Huella fiscal: ' + t.fiscalFingerprint);
      }
      btnRefreshCash.addEventListener('click', () => send(MESSAGE_TYPES.CASHIER_STATE, {}));
      btnZClosing.addEventListener('click', () => {
        if (!confirm('¿Confirmas el Cierre Z Diario? Cierra la jornada fiscal.')) return;
        setLoading(btnZClosing, true);
        send(MESSAGE_TYPES.Z_CLOSING, { confirm: true });
      });

      /* ── Fiscal ── */
      function renderFiscal(data) {
        if (!data) { fiscalResult.textContent = '—'; return; }
        fiscalResult.textContent = JSON.stringify(data, null, 2);
      }
      btnFiscalSummary.addEventListener('click', () => {
        const year = Number(fiscalYear.value), quarter = Number(fiscalQuarter.value);
        if (!year || !quarter) return warn('Indica año y trimestre.');
        setLoading(btnFiscalSummary, true);
        send(MESSAGE_TYPES.FISCAL_SUMMARY, { year, quarter });
      });
      btnFiscalBook.addEventListener('click', () => {
        const year = Number(fiscalYear.value), quarter = Number(fiscalQuarter.value);
        if (!year || !quarter) return warn('Indica año y trimestre.');
        setLoading(btnFiscalBook, true);
        send(MESSAGE_TYPES.FISCAL_BOOK, { year, quarter });
      });

      /* ── Documentos ── */
      function renderDocuments(items) {
        documentsBody.innerHTML = '';
        if (!Array.isArray(items) || items.length === 0) {
          documentsBody.innerHTML = '<tr><td colspan="4" style="text-align:center;color:var(--muted);">Sin paquetes preparados</td></tr>';
          return;
        }
        items.forEach((d) => {
          const tr = document.createElement('tr');
          [d.id || d._id || '—', d.periodLabel || d.periodType || '—', d.packageStatus || '—'].forEach((v) => {
            const td = document.createElement('td'); td.textContent = String(v); tr.appendChild(td);
          });
          const action = document.createElement('td');
          const b = document.createElement('button');
          b.className = 'btn btn-outline';
          b.style.cssText = 'width:auto;padding:5px 8px;margin:0;';
          b.textContent = 'Descargar';
          b.addEventListener('click', () => {
            setLoading(b, true);
            send(MESSAGE_TYPES.DOCUMENT_DOWNLOAD, { documentId: d.id || d._id });
          });
          action.appendChild(b); tr.appendChild(action); documentsBody.appendChild(tr);
        });
      }
      btnLoadPrepared.addEventListener('click', () => { setLoading(btnLoadPrepared, true); send(MESSAGE_TYPES.DOCUMENT_PREPARED, {}); });
      btnCreateSnapshot.addEventListener('click', () => {
        const year = Number(fiscalYear.value);
        const quarter = Number(fiscalQuarter.value);
        if (!Number.isInteger(year) || year < 2000 || year > 9999 ||
            !Number.isInteger(quarter) || quarter < 1 || quarter > 4) {
          err('Selecciona un ejercicio y trimestre válidos en el panel fiscal.');
          return;
        }
        if (!window.confirm('Se creará una nueva versión inmutable del resumen y libro fiscal del trimestre. No se enviará ningún correo. ¿Continuar?')) return;
        const periodKey = year + '-T' + quarter;
        if (!activeSnapshotOperationId || activeSnapshotPeriod !== periodKey) {
          activeSnapshotOperationId = createOperationUuid();
          activeSnapshotPeriod = periodKey;
        }
        if (!activeSnapshotOperationId) {
          err('El navegador no permite generar una clave segura de operación. No se ha creado ningún snapshot.');
          return;
        }
        setLoading(btnCreateSnapshot, true);
        send(MESSAGE_TYPES.CREATE_SNAPSHOT, {
          periodType: 'QUARTERLY',
          year,
          quarter,
          operationId: activeSnapshotOperationId
        });
      });

      /* ── Contexto ── */
      function applyContext(ctx) {
        contextData = ctx || {};
        const memberName = contextData.memberName || 'Marian';
        const roleLabel = contextData.roleLabel || 'PROPIETARIA';
        userInfo.textContent = `Sesion activa: ${memberName} | ${roleLabel}`;
        roleBadge.textContent = roleLabel;

        if (contextData.cashierState) renderCashier(contextData.cashierState);
        if (contextData.cashierMode) { /* badge opcional */ }

        // Fechas por defecto
        const today = contextData.today || new Intl.DateTimeFormat('sv-SE', { timeZone: contextData.timeZone || 'Europe/Madrid' }).format(new Date());
        const year = Number(today.slice(0, 4));
        fiscalYear.value = year;

        // Visibilidad por flags
        const flags = {
          cash: contextData.enableCashier !== false,
          fiscal: contextData.enableFiscal !== false,
          documents: contextData.enableDocuments !== false
        };
        $('cashPanel').style.display = flags.cash ? 'block' : 'none';
        $('fiscalPanel').style.display = flags.fiscal ? 'block' : 'none';
        $('documentsPanel').style.display = flags.documents ? 'block' : 'none';

        // Cargas iniciales
        if (flags.cash) send(MESSAGE_TYPES.CASHIER_STATE, {});

        if (fallbackTimer) { clearTimeout(fallbackTimer); fallbackTimer = null; }
        clearBanners();
      }

      /* ── Respuestas ── */
      function handleResponse(msg) {
        const type = String(msg.type || '').toUpperCase();
        const payload = msg.payload || {};
        const data = payload.data, error = payload.error, status = payload.status;
        const ok = status === 'SUCCESS';

        switch (type) {
          case 'CASHIER_STATE_RES':
            if (ok) renderCashier(data);
            break;
          case 'Z_CLOSING_RES':
            setLoading(btnZClosing, false);
            if (ok) { info('Cierre Z Diario completado.'); renderTicket(data?.ticket || data); send(MESSAGE_TYPES.CASHIER_STATE, {}); }
            else err(error?.message || 'Error en Cierre Z.');
            break;
          case 'FISCAL_SUMMARY_RES':
            setLoading(btnFiscalSummary, false);
            if (ok) renderFiscal(data);
            else err(error?.message || 'Error al obtener resumen fiscal.');
            break;
          case 'FISCAL_BOOK_RES':
            setLoading(btnFiscalBook, false);
            if (ok) renderFiscal(data);
            else err(error?.message || 'Error al obtener libro registro.');
            break;
          case 'DOCUMENT_PREPARED_RES':
            setLoading(btnLoadPrepared, false);
            if (ok) renderDocuments(data?.items || data || []);
            else err(error?.message || 'No se pudieron cargar los paquetes.');
            break;
          case 'CREATE_SNAPSHOT_RES':
            setLoading(btnCreateSnapshot, false);
            if (ok) {
              activeSnapshotOperationId = null;
              activeSnapshotPeriod = null;
              info('Snapshot inmutable creado: ' + (data?.documentId || 'versión registrada') + '. No se ha enviado a gestoría.');
              setLoading(btnLoadPrepared, true);
              send(MESSAGE_TYPES.DOCUMENT_PREPARED, {});
            } else {
              err(error?.message || 'No se pudo crear el snapshot. Conserva la misma operación al reintentar.');
            }
            break;
          case 'DOCUMENT_DOWNLOAD_RES':
            if (ok && data?.csvContent) {
              const id = String(data.documentId || 'paquete-fiscal').replace(/[^a-zA-Z0-9_-]/g, '_');
              downloadTextFile(id + '.csv', data.csvContent, 'text/csv;charset=utf-8');
              info('CSV descargado: ' + id);
            } else if (ok && data?.summaryText) {
              const id = String(data.documentId || 'paquete-fiscal').replace(/[^a-zA-Z0-9_-]/g, '_');
              downloadTextFile(id + '-resumen.txt', data.summaryText, 'text/plain;charset=utf-8');
              info('Resumen descargado: ' + id);
            } else if (!ok) err(error?.message || 'No se pudo descargar.');
            else warn('El paquete no contiene contenido descargable.');
            break;
        }
      }

      window.addEventListener('message', (e) => {
        const msg = e?.data || {};
        const type = String(msg.type || '').trim().toUpperCase();
        if (!type || !VALID_TYPES.has(type)) return;
        if (e.source !== window.parent) return;
        if (e.origin && e.origin !== 'null') PARENT_ORIGIN = e.origin;
        if (type === MESSAGE_TYPES.READY) return;
        if (type === MESSAGE_TYPES.CONTEXT) { applyContext(msg.payload || {}); return; }
        handleResponse(msg);
      });

      send(MESSAGE_TYPES.READY, { traceId });
      setTimeout(() => { if (!contextData) send(MESSAGE_TYPES.READY, { traceId }); }, 600);
      setTimeout(() => { if (!contextData) send(MESSAGE_TYPES.READY, { traceId }); }, 1800);

      fallbackTimer = setTimeout(() => {
        if (!contextData) {
          console.warn('[admin-widget] MM_CONTEXT no recibido, aplicando defaults');
          // Fallback fail-closed: no activar funciones que dependen de un
          // contexto/contrato backend que no ha llegado.
          applyContext({
            roleLabel: 'CONTEXTO NO DISPONIBLE',
            enableClocking: false,
            enableCashier: false,
            enableInventory: false,
            enableScheduleQuery: false,
            enableFiscal: false,
            enableDocuments: false,
            timeZone: 'Europe/Madrid',
            currencyCode: 'EUR'
          });
          warn('No se ha recibido el contexto autorizado desde Velo. Las funciones quedan deshabilitadas.');
        }
      }, 3000);
    })();
  </script>
