        if (!Number.isFinite(n)) return '--';
        return new Intl.NumberFormat('es-ES',{style:'currency',currency:contextData?.currencyCode||'EUR'}).format(n);
      }

      /* ── AI ── */
      function appendAIMessage(role, text) {
        const el = document.createElement('div');
        el.className = 'ai-msg ' + role;
        el.textContent = text;
        aiChat.appendChild(el);
        aiChat.scrollTop = aiChat.scrollHeight;
        return el;
      }
      function sendAIQuery() {
        const q = (aiInput.value || '').trim();
        if (!q) return warn('Escribe una pregunta para la IA.');
        appendAIMessage('user', q);
        aiInput.value = '';
        const loading = appendAIMessage('loading', 'Pensando…');
        setLoading(btnAISend, true);
        send(MESSAGE_TYPES.AI_QUERY, { query: q, conversationId });
        loading.dataset.pending = '1';
      }
      btnAISend.addEventListener('click', sendAIQuery);
      aiInput.addEventListener('keydown', (e) => { if (e.key === 'Enter') sendAIQuery(); });

      /* ── Destinatarios ── */
      function renderRecipients(items) {
        recipientSelect.innerHTML = '';
        if (!Array.isArray(items) || items.length === 0) {
          recipientSelect.innerHTML = '<option value="">Sin destinatarios configurados</option>';
          return;
        }
        items.forEach((r) => {
          const opt = document.createElement('option');
          opt.value = r.id || r.recipientId;
          opt.textContent = r.label || r.email || r.name || opt.value;
          recipientSelect.appendChild(opt);
        });
      }

      /* ── Envío ── */

        formMonthly.style.display = which === 'monthly' ? 'block' : 'none';
        formQuarterly.style.display = which === 'quarterly' ? 'block' : 'none';
        formSpecific.style.display = which === 'specific' ? 'block' : 'none';
      }

      $('btnOpenQuarterly').addEventListener('click', () => toggleForms('quarterly'));

      $('btnCancelMonthly').addEventListener('click', () => toggleForms(null));
      $('btnCancelQuarterly').addEventListener('click', () => toggleForms(null));
      $('btnCancelSpecific').addEventListener('click', () => toggleForms(null));

      $('btnConfirmMonthly').addEventListener('click', () => {
        const recipientId = recipientSelect.value;

        const month = Number(monthlyMonth.value);
        if (!recipientId) return warn('Selecciona un destinatario.');

      });
      $('btnConfirmQuarterly').addEventListener('click', () => {
        const recipientId = recipientSelect.value;
        const year = Number(quarterlyYear.value);

        if (!recipientId) return warn('Selecciona un destinatario.');
        if (!year || !quarter) return warn('Indica año y trimestre.');
        setLoading($('btnConfirmQuarterly'), true);
        send(MESSAGE_TYPES.SEND_QUARTERLY_PACKAGE, { recipientId, year, quarter, notes: quarterlyNotes.value.trim() || null });
      });
      $('btnConfirmSpecific').addEventListener('click', () => {
        const recipientId = recipientSelect.value;
        const documentId = specificDocId.value.trim();
        if (!recipientId) return warn('Selecciona un destinatario.');
        if (!documentId) return warn('Indica el ID del documento.');
        setLoading($('btnConfirmSpecific'), true);
        send(MESSAGE_TYPES.SEND_SPECIFIC_DOCUMENT, {
          recipientId, documentId,
          documentType: specificDocType.value || null,
          notes: specificNotes.value.trim() || null
        });
      });

      /* ── Staff ── */
      function fillStaffSelects(items) {
        hoursStaffSelect.innerHTML = '<option value="">—</option>';
        notifyStaffSelect.innerHTML = '<option value="">—</option>';
        items.forEach((s) => {
          const value = s.id || s.employeeId;
          const label = s.name || s.displayName || value;
          [hoursStaffSelect, notifyStaffSelect].forEach((sel) => {
            const opt = document.createElement('option');
            opt.value = value; opt.textContent = label;
            sel.appendChild(opt);
          });
        });
      }
      function renderStaff(items) {
        staffCache = items || [];
        if (!staffCache.length) {
          staffBody.innerHTML = '<tr><td colspan="4" style="text-align:center;color:var(--muted);">Sin empleados</td></tr>';
          fillStaffSelects([]); return;
        }
        staffBody.innerHTML = '';
        staffCache.forEach((s) => {
          const tr = document.createElement('tr');
          [s.name || s.displayName || '—', s.roleLabel || s.role || '—', s.email || '—', s.status || 'activo'].forEach((v) => {
            const td = document.createElement('td'); td.textContent = String(v); tr.appendChild(td);
          });
          staffBody.appendChild(tr);
        });
        fillStaffSelects(staffCache);
      }
      btnLoadStaff.addEventListener('click', () => { setLoading(btnLoadStaff, true); send(MESSAGE_TYPES.STAFF_LIST, {}); });
      btnLoadStaffHours.addEventListener('click', () => {
        const employeeId = hoursStaffSelect.value;
        const startDateYMD = staffHoursStart.value;
        const endDateYMD = staffHoursEnd.value;
        if (!employeeId || !startDateYMD || !endDateYMD) return warn('Completa empleado, desde y hasta.');
        setLoading(btnLoadStaffHours, true);
        send(MESSAGE_TYPES.STAFF_HOURS_SUMMARY, { employeeId, startDateYMD, endDateYMD });
      });
      btnNotifyStaff.addEventListener('click', () => {
        const employeeId = notifyStaffSelect.value;
        const subject = notifySubject.value.trim();
        const message = notifyMessage.value.trim();
        if (!employeeId || !subject || !message) return warn('Completa empleado, asunto y mensaje.');
        setLoading(btnNotifyStaff, true);
        send(MESSAGE_TYPES.STAFF_NOTIFY, {
          employeeId, channel: notifyChannel.value || 'EMAIL', subject, message
        });
      });

      /* ── Caja ── */
      function renderCashier(state) {
        if (!state) { cashSaldo.textContent='--'; cashOps.textContent='--'; cashDiaKey.textContent='--'; return; }
        cashSaldo.textContent = money(state.saldoTotal);
        cashOps.textContent = Number.isFinite(Number(state.totalOperaciones)) ? String(state.totalOperaciones) : '--';
        cashDiaKey.textContent = state.diaKey || '--';
      }
      function renderTicket(t) {
        if (!t) { cashTicket.style.display='none'; return; }
        cashTicket.style.display = 'block';
        const rows = [];
        rows.push(`<div style="text-align:center;font-weight:900;text-transform:uppercase">${t.title || 'TICKET'}</div>`);
        if (t.ticketNumber) rows.push(`Nº: ${t.ticketNumber}`);
        if (t.concept) rows.push(`Concepto: ${t.concept}`);
        if (t.amount != null) rows.push(`Importe: ${money(t.amount)}`);
        if (t.traceId) rows.push(`Traza: ${t.traceId}`);
        if (t.fiscalFingerprint) rows.push(`Huella fiscal: ${t.fiscalFingerprint}`);
        cashTicket.innerHTML = rows.join('<br/>');
      }
      btnRefreshCash.addEventListener('click', () => send(MESSAGE_TYPES.CASHIER_STATE, {}));
      btnZClosing.addEventListener('click', () => {
        if (!confirm('¿Confirmas el Cierre Z Diario? Cierra la jornada fiscal.')) return;
        setLoading(btnZClosing, true);
        send(MESSAGE_TYPES.Z_CLOSING, { confirm: true });
      });

      /* ── Fiscal ── */
      function renderFiscal(data) {
        if (!data) { fiscalResult.textContent = '—'; return; }
        fiscalResult.textContent = JSON.stringify(data, null, 2);
      }
      btnFiscalSummary.addEventListener('click', () => {
        const year = Number(fiscalYear.value), quarter = Number(fiscalQuarter.value);
        if (!year || !quarter) return warn('Indica año y trimestre.');
        setLoading(btnFiscalSummary, true);
        send(MESSAGE_TYPES.FISCAL_SUMMARY, { year, quarter });
      });
      btnFiscalBook.addEventListener('click', () => {
        const year = Number(fiscalYear.value), quarter = Number(fiscalQuarter.value);
        if (!year || !quarter) return warn('Indica año y trimestre.');
        setLoading(btnFiscalBook, true);
        send(MESSAGE_TYPES.FISCAL_BOOK, { year, quarter });
      });

      /* ── Documentos ── */
      function renderDocuments(items) {
        documentsBody.innerHTML = '';
        if (!Array.isArray(items) || items.length === 0) {
          documentsBody.innerHTML = '<tr><td colspan="4" style="text-align:center;color:var(--muted);">Sin paquetes preparados</td></tr>';
          return;
        }
        items.forEach((d) => {
          const tr = document.createElement('tr');
          [d.id || d._id || '—', d.periodLabel || d.periodType || '—', d.status || '—'].forEach((v) => {
            const td = document.createElement('td'); td.textContent = String(v); tr.appendChild(td);
          });
          const action = document.createElement('td');
          const b = document.createElement('button');
          b.className = 'btn btn-outline';
          b.style.cssText = 'width:auto;padding:5px 8px;margin:0;';
          b.textContent = 'Descargar';
          b.addEventListener('click', () => {
            setLoading(b, true);
            send(MESSAGE_TYPES.DOCUMENT_DOWNLOAD, { documentId: d.id || d._id });
          });
          action.appendChild(b); tr.appendChild(action); documentsBody.appendChild(tr);
        });
      }
      btnLoadPrepared.addEventListener('click', () => { setLoading(btnLoadPrepared, true); send(MESSAGE_TYPES.DOCUMENT_PREPARED, {}); });

      /* ── Contexto ── */
      function applyContext(ctx) {
        contextData = ctx || {};
        const memberName = contextData.memberName || 'Marian';
        const roleLabel = contextData.roleLabel || 'PROPIETARIA';
        userInfo.textContent = `Sesion activa: ${memberName} | ${roleLabel}`;
        roleBadge.textContent = roleLabel;

        if (contextData.cashierState) renderCashier(contextData.cashierState);
        if (contextData.cashierMode) { /* badge opcional */ }

        // Fechas por defecto
        const today = contextData.today || new Intl.DateTimeFormat('sv-SE', { timeZone: contextData.timeZone || 'Europe/Madrid' }).format(new Date());
        const year = Number(today.slice(0, 4));
        monthlyYear.value = year; quarterlyYear.value = year; fiscalYear.value = year;
        staffHoursStart.value = today; staffHoursEnd.value = today;

        // Visibilidad por flags
        const flags = {
          ai: contextData.enableAIAssistant !== false,
          send: contextData.enableAccountantSend !== false,
          staff: contextData.enableStaffManagement !== false,
          cash: contextData.enableCashier !== false,
          fiscal: contextData.enableFiscal !== false,
          documents: contextData.enableDocuments !== false
        };
        $('aiPanel').style.display = flags.ai ? 'block' : 'none';
        $('sendPanel').style.display = flags.send ? 'block' : 'none';
        $('staffPanel').style.display = flags.staff ? 'block' : 'none';
        $('cashPanel').style.display = flags.cash ? 'block' : 'none';
        $('fiscalPanel').style.display = flags.fiscal ? 'block' : 'none';
        $('documentsPanel').style.display = flags.documents ? 'block' : 'none';

        // Cargas iniciales
        if (flags.send) send(MESSAGE_TYPES.ACCOUNTANT_RECIPIENTS, {});
        if (flags.cash) send(MESSAGE_TYPES.CASHIER_STATE, {});

        if (fallbackTimer) { clearTimeout(fallbackTimer); fallbackTimer = null; }
        clearBanners();
      }

      /* ── Respuestas ── */
      function handleResponse(msg) {
        const type = String(msg.type || '').toUpperCase();
        const payload = msg.payload || {};
        const data = payload.data, error = payload.error, status = payload.status;
        const ok = status === 'SUCCESS';

        switch (type) {
          case 'AI_QUERY_RES': {
            setLoading(btnAISend, false);
            const pending = aiChat.querySelector('.ai-msg.loading[data-pending="1"]');
            if (pending) pending.remove();
            if (ok && data) {
              if (data.conversationId) conversationId = data.conversationId;
              const text = data.text || data.message || data.answer || '(sin respuesta)';
              appendAIMessage('assistant', text);
            } else {
              appendAIMessage('error', error?.message || 'Error al consultar la IA.');
            }
            break;
          }
          case 'ACCOUNTANT_RECIPIENTS_RES':
            if (ok) renderRecipients(data?.items || data?.recipients || data);
            else warn(error?.message || 'No se pudieron cargar los destinatarios.');
            break;
          case 'SEND_MONTHLY_PACKAGE_RES':
            setLoading($('btnConfirmMonthly'), false);
            if (ok) { info('Paquete mensual enviado al gestor.'); toggleForms(null); }
            else err(error?.message || 'No se pudo enviar el paquete mensual.');
            break;
          case 'SEND_QUARTERLY_PACKAGE_RES':
            setLoading($('btnConfirmQuarterly'), false);
            if (ok) { info('Paquete trimestral enviado al gestor.'); toggleForms(null); }
            else err(error?.message || 'No se pudo enviar el paquete trimestral.');
            break;
          case 'SEND_SPECIFIC_DOCUMENT_RES':
            setLoading($('btnConfirmSpecific'), false);
            if (ok) { info('Documento enviado al gestor.'); toggleForms(null); }
            else err(error?.message || 'No se pudo enviar el documento.');
            break;
          case 'STAFF_LIST_RES':
            setLoading(btnLoadStaff, false);
            if (ok) renderStaff(data?.items || data?.employees || data || []);
            else err(error?.message || 'No se pudo cargar el equipo.');
            break;
          case 'STAFF_HOURS_SUMMARY_RES':
            setLoading(btnLoadStaffHours, false);
            if (ok && data) {
              const h = Number(data.horasTrabajadasDecimal || data.totalHoras || 0);
              staffHoursResult.textContent = `Horas trabajadas: ${h}h` +
                (data.horasPausaDecimal != null ? ` | Pausas: ${Number(data.horasPausaDecimal)}h` : '') +
                (data.totalFichajesProcesados != null ? ` | Fichajes: ${Number(data.totalFichajesProcesados)}` : '');
            } else staffHoursResult.textContent = error?.message || 'Sin datos.';
            break;
          case 'STAFF_NOTIFY_RES':
            setLoading(btnNotifyStaff, false);
            if (ok) { info('Aviso enviado.'); notifySubject.value=''; notifyMessage.value=''; }
            else err(error?.message || 'No se pudo enviar el aviso.');
            break;
          case 'CASHIER_STATE_RES':
            if (ok) renderCashier(data);
            break;
          case 'Z_CLOSING_RES':
            setLoading(btnZClosing, false);
            if (ok) { info('Cierre Z Diario completado.'); renderTicket(data?.ticket || data); send(MESSAGE_TYPES.CASHIER_STATE, {}); }
            else err(error?.message || 'Error en Cierre Z.');
            break;
          case 'FISCAL_SUMMARY_RES':
            setLoading(btnFiscalSummary, false);
            if (ok) renderFiscal(data);
            else err(error?.message || 'Error al obtener resumen fiscal.');
            break;
          case 'FISCAL_BOOK_RES':
            setLoading(btnFiscalBook, false);
            if (ok) renderFiscal(data);
            else err(error?.message || 'Error al obtener libro registro.');
            break;
          case 'DOCUMENT_PREPARED_RES':
            setLoading(btnLoadPrepared, false);
            if (ok) renderDocuments(data?.items || data || []);
            else err(error?.message || 'No se pudieron cargar los paquetes.');
            break;
          case 'DOCUMENT_DOWNLOAD_RES':
            if (ok && data?.url) window.open(data.url, '_blank');
            else if (!ok) err(error?.message || 'No se pudo descargar.');
            break;
          case MESSAGE_TYPES.ONLY_STAFF_ERROR:
          case MESSAGE_TYPES.STAFF_ERROR:
            err(error?.message || 'Error en la solicitud.');
            break;
        }
      }

      window.addEventListener('message', (e) => {
        const msg = e?.data || {};
        const type = String(msg.type || '').trim().toUpperCase();
        if (!type || !VALID_TYPES.has(type)) return;
        if (e.source !== window.parent) return;
        if (e.origin && e.origin !== 'null') PARENT_ORIGIN = e.origin;
        if (type === MESSAGE_TYPES.READY) return;
        if (type === MESSAGE_TYPES.CONTEXT) { applyContext(msg.payload || {}); return; }
        handleResponse(msg);
      });

      send(MESSAGE_TYPES.READY, { traceId });
      setTimeout(() => { if (!contextData) send(MESSAGE_TYPES.READY, { traceId }); }, 600);
      setTimeout(() => { if (!contextData) send(MESSAGE_TYPES.READY, { traceId }); }, 1800);

      fallbackTimer = setTimeout(() => {
        if (!contextData) {
          console.warn('[admin-widget] MM_CONTEXT no recibido, aplicando defaults');
          applyContext({
            roleLabel: 'PROPIETARIA',
            enableClocking: false,
            enableCashier: true,
            enableInventory: true,
            enableScheduleQuery: true,
            enableStaffManagement: true,
            enableAIAssistant: true,
            enableAccountantSend: true,
            enableFiscal: true,
            enableDocuments: true,
            timeZone: 'Europe/Madrid',
            currencyCode: 'EUR'
          });
        }
      }, 3000);
    })();
  </script>
