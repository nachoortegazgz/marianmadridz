      /* Loading con timeout (evita quedar bloqueado si no llega el _RES) */
      function _setLoading(btn, loading, timeoutMs) {
        if (!btn) return;
        if (loading) {
          if (!btn.dataset.original) btn.dataset.original = btn.textContent;
          btn.disabled = true; btn.textContent = 'Procesando...';
          const token = (btn._loadToken || 0) + 1;
          btn._loadToken = token;
          if (timeoutMs) {
            setTimeout(() => {
              if (btn._loadToken === token && btn.disabled) {
                btn.disabled = false;
                btn.textContent = btn.dataset.original || btn.textContent;
                _showWarn('Operación sin respuesta del servidor. Reintenta o revisa el bridge.');
              }
            }, timeoutMs);
          }
        } else {
          btn._loadToken = (btn._loadToken || 0) + 1;
          btn.disabled = false; btn.textContent = btn.dataset.original || btn.textContent;
        }
      }

      function _formatMadridDate(value, dateOnly) {
        if (!value) return '';
        const date = new Date(value);
        if (Number.isNaN(date.getTime())) return '';
        const timeZone = contextData?.timeZone || 'Europe/Madrid';
        const options = dateOnly
          ? { year: 'numeric', month: '2-digit', day: '2-digit', timeZone }
          : { year: 'numeric', month: '2-digit', day: '2-digit', hour: '2-digit', minute: '2-digit', timeZone };
        return new Intl.DateTimeFormat('es-ES', options).format(date);
      }
      function _formatMoney(value) {
        const amount = Number(value);
        if (!Number.isFinite(amount)) return '--';
        return new Intl.NumberFormat('es-ES', {
          style: 'currency', currency: contextData?.currencyCode || 'EUR',
          minimumFractionDigits: 2, maximumFractionDigits: 2
        }).format(amount);
      }

      /* ── Estado de jornada ── */
      function _updateStatus(state) {
        if (!state) return;
        const estado = state.estadoActual || 'INACTIVO';
        const map = {
          TRABAJANDO: { text: 'En jornada', cls: 'activo' },
          EN_PAUSA: { text: 'En pausa', cls: 'pausa' },
          INACTIVO: { text: 'Inactivo', cls: 'inactivo' },
          AJUSTADO: { text: 'Ajustado', cls: 'inactivo' }
        };
        const info = map[estado] || { text: estado, cls: 'inactivo' };
        statusValue.textContent = info.text;
        statusValue.className = `status-value ${info.cls}`;
        statusBox.dataset.status = estado;
        if (state.ultimoFichaje) {
          const uf = state.ultimoFichaje;
          statusTime.textContent = `Ultimo fichaje: ${(uf.tipo || uf.tipoFichaje || '').trim()} ${_formatMadridDate(uf.fechaHora, false)}`.trim();
        } else { statusTime.textContent = 'Sin fichajes registrados'; }
        btnEntrada.disabled = estado === 'TRABAJANDO' || estado === 'EN_PAUSA';
        btnSalida.disabled = estado !== 'TRABAJANDO' && estado !== 'EN_PAUSA';
        btnPausaInicio.disabled = estado !== 'TRABAJANDO';
        btnPausaFin.disabled = estado !== 'EN_PAUSA';
      }

      /* ── Caja ── */
      function _renderCashierState(state) {
        if (!state || typeof state !== 'object') {
          cashSaldo.textContent = '--'; cashOps.textContent = '--';
          cashDiaKey.textContent = '--'; cashMode.textContent = contextData?.cashierMode || '--';
          _renderMovements([]);
          return;
        }
        // DTO canónico de cajas.web.js:getCashierState.
        cashSaldo.textContent = _formatMoney(state.netAmount);
        cashOps.textContent = Number.isFinite(Number(state.totalOperations)) ? String(state.totalOperations) : '--';
        cashDiaKey.textContent = state.operationDate || '--';
        cashMode.textContent = contextData?.cashierMode || state.cashierMode || '--';
        const movements = Array.isArray(state.movimientos) ? state.movimientos
          : Array.isArray(state.recentMovements) ? state.recentMovements : [];
        _renderMovements(movements);
      }

      function _renderMovements(items) {
        const movements = Array.isArray(items) ? items : [];
        if (!movements.length) {
          movementsBody.innerHTML = '<tr><td colspan="5" style="text-align:center; color:var(--muted);">Sin movimientos</td></tr>';
          return;
        }
        movementsBody.innerHTML = '';
        movements.forEach((m) => {
          const tr = document.createElement('tr');
          const hora = m.hora || (m.fechaHora ? new Intl.DateTimeFormat('es-ES', { hour: '2-digit', minute: '2-digit', timeZone: contextData?.timeZone || 'Europe/Madrid' }).format(new Date(m.fechaHora)) : '');
          const tipo = m.tipo || m.txType || '';
          const concepto = m.concepto || m.concept || '';
          const importe = m.importe != null ? _formatMoney(m.importe) : (m.amount != null ? _formatMoney(m.amount) : '');
          const traza = m.traza || m.traceId || m.huella || m.fiscalFingerprint || '';
          [hora, tipo, concepto, importe, traza].forEach((v) => {
            const td = document.createElement('td'); td.textContent = String(v); tr.appendChild(td);
          });
          movementsBody.appendChild(tr);
        });
      }

      function _renderTicket(ticket) {
        if (!ticket) { ticketDisplay.style.display = 'none'; ticketBox.replaceChildren(); return; }
        ticketDisplay.style.display = 'block';
        ticketBox.replaceChildren();
        const appendText = (tag, className, value) => {
          const element = document.createElement(tag);
          if (className) element.className = className;
          element.textContent = String(value);
          ticketBox.appendChild(element);
          return element;
        };
        const appendRow = (label, value, mono = false) => {
          const row = document.createElement('div');
          row.className = 'ticket-row';
          const key = document.createElement('span');
          key.textContent = label;
          const val = document.createElement('span');
          if (mono) val.className = 'ticket-mono';
          val.textContent = String(value);
          row.append(key, val);
          ticketBox.appendChild(row);
        };
        appendText('div', 'ticket-title', ticket.title || 'TICKET');
        appendRow('Nº:', ticket.ticketNumber || '—');
        appendRow('Fecha:', _formatMadridDate(ticket.createdAt || new Date(), false));
        if (ticket.supplierName) appendRow('Proveedor:', ticket.supplierName);
        if (ticket.supplierNif) appendRow('NIF:', ticket.supplierNif);
        if (ticket.concept) appendRow('Concepto:', ticket.concept);
        if (ticket.amount != null) appendRow('Importe:', _formatMoney(ticket.amount));
        if (ticket.paymentMethod) appendRow('Método:', ticket.paymentMethod);
        appendText('div', 'ticket-sep', '');
        if (ticket.traceId) appendRow('Traza:', ticket.traceId, true);
        if (ticket.fiscalFingerprint) appendRow('Huella fiscal:', ticket.fiscalFingerprint, true);
        if (ticket.issuerName) {
          appendText('div', 'ticket-sep', '');
          appendText('div', 'ticket-issuer', ticket.issuerName);
        }
        if (ticket.issuerCif) appendText('div', 'ticket-issuer', 'CIF: ' + ticket.issuerCif);
      }

      /* ── Historial / Horas ── */
      function _renderHistory(items) {
        historyBody.innerHTML = '';
        if (!items || items.length === 0) {
          historyBody.innerHTML = '<tr><td colspan="3" style="text-align:center; color:var(--muted);">No hay fichajes en este periodo</td></tr>';
          return;
        }
        items.forEach((item) => {
          const tr = document.createElement('tr');
          const fecha = item.fechaHora ? _formatMadridDate(item.fechaHora, true) : item.diaKey || '';
          const hora = item.hora || (item.fechaHora ? new Intl.DateTimeFormat('es-ES', { hour: '2-digit', minute: '2-digit', timeZone: contextData?.timeZone || 'Europe/Madrid' }).format(new Date(item.fechaHora)) : '');
          const tipoBase = item.tipo || item.tipoFichaje || '';
          const integrityLabel = item.integrityStatus === 'UNVERIFIED'
            ? ' · NO VERIFICADO'
            : (item.sequenceStatus === 'INVALID' ? ' · SECUENCIA INVÁLIDA' : '');
          const tipo = tipoBase + integrityLabel;
          [fecha, hora, tipo].forEach((value) => {
            const td = document.createElement('td'); td.textContent = String(value); tr.appendChild(td);
          });
          historyBody.appendChild(tr);
        });
      }

      function _renderHoursResult(data) {
        hoursResult.replaceChildren();
        const appendLine = (value, style) => {
          const line = document.createElement('div');
          if (style) line.style.cssText = style;
          line.textContent = String(value);
          hoursResult.appendChild(line);
        };
        if (!data) {
          quickHours.textContent = 'Consultar';
          appendLine('Horas trabajadas: —', 'color:var(--muted)');
          appendLine('Tiempo de pausa: —', 'color:var(--muted)');
          return;
        }
        const workedHours = Number(data.horasTrabajadasDecimal || 0);
        const formatted = String(data.tiempoFormateado || '');
        quickHours.textContent = formatted || `${workedHours}h`;
        appendLine(`Horas trabajadas: ${workedHours}h (${formatted})`, 'font-weight:600');
        appendLine(`Tiempo de pausa: ${Number(data.horasPausaDecimal || 0)}h`, 'color:var(--muted)');
        appendLine(`Fichajes procesados: ${Number(data.totalFichajesProcesados || 0)}`, 'font-size:12px;color:var(--muted)');
        if (Number(data.jornadasAbiertas || 0) > 0) {
          appendLine(`Jornadas abiertas: ${Number(data.jornadasAbiertas)} (el tiempo en curso no se incluye)`, 'font-size:12px;color:var(--muted)');
        }
      }

      /* ═════════════════════════════════════════════════════════════════
         CONTEXTO — alineado con zonastaff.js v5004
         Seguridad: no se muestra ninguna operación sin contexto autorizado.
         ═════════════════════════════════════════════════════════════════ */
      function _renderContext(ctx) {
        contextData = ctx || {};
        if (contextData.isAuthorized !== true) {
          [clockPanel, historyPanel, hoursPanel, cashPanel, adminPanel]
            .forEach((panel) => { if (panel) panel.style.display = 'none'; });
          _showError('Acceso no autorizado o contexto no disponible.');
          return;
        }

        /* Cabecera */
        const memberName = contextData.memberName || contextData.employeeName || 'EMPLEADO';
        const roleLabel = contextData.roleLabel || (contextData.isAdmin ? 'ADMINISTRACION' : (contextData.isCajero ? 'CAJA' : 'EQUIPO'));
        const panelType = contextData.panelType || 'TPV';
        userInfo.textContent = `Sesion activa: ${memberName} | ${roleLabel} | ${panelType}`;
        roleBadge.textContent = roleLabel;

        /* Feature flags: fail closed; the backend must explicitly enable them. */
        const enableClocking      = contextData.enableClocking      === true;
        const enableBreaks        = contextData.enableBreaks        === true;
        const enableCashier       = contextData.enableCashier       === true;
        const enableScheduleQuery = contextData.enableScheduleQuery === true;  // default OFF
        const isAdmin   = contextData.isAdmin === true;

        /* Visibilidad */
        clockPanel.style.display     = enableClocking ? 'block' : 'none';
        historyPanel.style.display   = enableClocking ? 'block' : 'none';
        hoursPanel.style.display     = enableScheduleQuery ? 'block' : 'none';
        cashPanel.style.display      = enableCashier ? 'block' : 'none';
        adminPanel.style.display     = isAdmin ? 'block' : 'none';

        btnPausaInicio.style.display = enableBreaks ? '' : 'none';
        btnPausaFin.style.display    = enableBreaks ? '' : 'none';

        /* Caja: cabecera y estado inicial */
        if (enableCashier) {
          cashMode.textContent = contextData.cashierMode || '—';
          cashModeBadge.textContent = contextData.cashierMode ? `(${contextData.cashierMode})` : '';
          if (contextData.cashierState) _renderCashierState(contextData.cashierState);
          setTimeout(() => _sendToVelo(MESSAGE_TYPES.CASHIER_STATE, {}), 700);
        }

        /* Admin: opciones de empleados */
        if (isAdmin && Array.isArray(contextData.staffOptions)) {
          adjEmployee.innerHTML = '';
          contextData.staffOptions.forEach((s) => {
            const opt = document.createElement('option');
            opt.value = s.id || s.value;
            opt.textContent = s.name || s.label || String(s);
            adjEmployee.appendChild(opt);
          });
        }

        /* Estado de jornada */
        if (contextData.estadoInicial) {
          _updateStatus(contextData.estadoInicial);
        } else if (enableClocking) {
          setTimeout(() => _sendToVelo(MESSAGE_TYPES.GET_STATE, {}), 500);
        }

        /* Fechas por defecto */
        const today = contextData.today || new Intl.DateTimeFormat('sv-SE', { timeZone: contextData.timeZone || 'Europe/Madrid' }).format(new Date());
        histStart.value = today; histEnd.value = today;
        hoursStart.value = today; hoursEnd.value = today;

        /* Cancelar fallback: contexto real ya llegó */
        if (contextFallbackTimer) { clearTimeout(contextFallbackTimer); contextFallbackTimer = null; }

        _hideAllBanners();
      }

      /* Fechas por defecto (antes de cualquier contexto) */
      const initialToday = new Intl.DateTimeFormat('sv-SE', { timeZone: 'Europe/Madrid' }).format(new Date());
      histStart.value = initialToday; histEnd.value = initialToday;
      hoursStart.value = initialToday; hoursEnd.value = initialToday;

      /* ── Tabs de caja ── */
      const cashTabs = document.querySelectorAll('.cash-tab');
      cashTabs.forEach((tab) => {
        tab.addEventListener('click', () => {
          cashTabs.forEach((t) => t.classList.remove('active'));
          tab.classList.add('active');
          tabSupplier.style.display = 'none';
          tabWithdrawal.style.display = 'none';
          tabReports.style.display = 'none';
          const target = tab.dataset.tab;
          if (target === 'supplier') tabSupplier.style.display = 'block';
          else if (target === 'withdrawal') tabWithdrawal.style.display = 'block';
          else if (target === 'reports') tabReports.style.display = 'block';
        });
      });

      /* ── Listeners fichaje ── */
      btnEntrada.addEventListener('click', () => _sendToVelo(MESSAGE_TYPES.CLOCK, { tipoFichaje: 'ENTRADA' }));
      btnSalida.addEventListener('click', () => _sendToVelo(MESSAGE_TYPES.CLOCK, { tipoFichaje: 'SALIDA' }));
      btnPausaInicio.addEventListener('click', () => _sendToVelo(MESSAGE_TYPES.CLOCK, { tipoFichaje: 'PAUSA_INICIO' }));
      btnPausaFin.addEventListener('click', () => _sendToVelo(MESSAGE_TYPES.CLOCK, { tipoFichaje: 'PAUSA_FIN' }));
      btnRefreshState.addEventListener('click', () => _sendToVelo(MESSAGE_TYPES.GET_STATE, {}));
      btnOpenCalendar.addEventListener('click', () => _sendToVelo(MESSAGE_TYPES.NAV, { target: 'STAFF_CALENDAR' }));

      /* ── Listeners historial / horas / admin ── */
      btnLoadHistory.addEventListener('click', () => {
        const start = histStart.value, end = histEnd.value;
        if (!start || !end) return _showWarn('Selecciona un rango de fechas valido.');
        _sendToVelo(MESSAGE_TYPES.GET_HISTORY, { startDateYMD: start, endDateYMD: end });
      });
      btnCalcHours.addEventListener('click', () => {
        const start = hoursStart.value, end = hoursEnd.value;
        if (!start || !end) return _showWarn('Selecciona un rango de fechas valido.');
        _setLoading(btnCalcHours, true);
        _sendToVelo(MESSAGE_TYPES.HOURS_CALC, { startDateYMD: start, endDateYMD: end });
      });
      btnAdjSubmit.addEventListener('click', () => {
        const employeeId = adjEmployee.value;
        const motivo = (adjMotivo.value || '').trim();
        if (!employeeId || !motivo) return _showWarn('Completa Empleado y Motivo del ajuste.');
        _setLoading(btnAdjSubmit, true);
        _sendToVelo(MESSAGE_TYPES.ADJUSTMENT, { employeeId, motivoAjuste: motivo });
      });

      /* ── Listeners caja (con timeout de 12s) ── */
      btnSupplierPayment.addEventListener('click', () => {
        const name = (supplierName.value || '').trim();
        const nif = (supplierNif.value || '').trim();
        const amount = Number(supplierAmount.value);
        const concept = (supplierConcept.value || '').trim();
        const invoiceRef = (supplierInvoiceRef.value || '').trim();
        if (!name || !concept || !Number.isFinite(amount) || amount <= 0) {
          _showWarn('Completa proveedor, concepto e importe válido.'); return;
        }
        _setLoading(btnSupplierPayment, true, 12000);
        _sendToVelo(MESSAGE_TYPES.TPV_TX, {
          txType: 'PAGO_PROVEEDOR',
          supplierName: name, supplierNif: nif, amount, concept,
          invoiceRef: invoiceRef || null, paymentMethod: 'EFECTIVO',
          generateTicket: true, requestFiscalFingerprint: true
        });
      });

      btnWithdrawal.addEventListener('click', () => {
        const amount = Number(withdrawalAmount.value);
        const concept = (withdrawalConcept.value || '').trim();
        if (!concept || !Number.isFinite(amount) || amount <= 0) {
          _showWarn('Completa concepto e importe válido.'); return;
        }
        _setLoading(btnWithdrawal, true, 12000);
        _sendToVelo(MESSAGE_TYPES.TPV_TX, {
          txType: 'SALIDA_EFECTIVO', amount, concept,
          generateTicket: true, requestFiscalFingerprint: true
        });
      });

      btnXCount.addEventListener('click', () => {
        _setLoading(btnXCount, true, 12000);
        _sendToVelo(MESSAGE_TYPES.X_COUNT, { includeBreakdown: true, requestFiscalFingerprint: true });
      });

      btnZClosing.addEventListener('click', () => {
        if (!confirm('¿Confirmas el Cierre Z Diario? Esta acción cierra la jornada fiscal.')) return;
        _setLoading(btnZClosing, true, 12000);
        _sendToVelo(MESSAGE_TYPES.Z_CLOSING, { confirm: true, requestFiscalFingerprint: true });
      });

      /* ── Respuestas ── */
      function _handleResponse(msg) {
        const type = String(msg.type || '').toUpperCase();
        const payload = msg.payload || {};
        const data = payload.data, error = payload.error, status = payload.status;

        switch (type) {
          case MESSAGE_TYPES.GET_STATE_RES:
            if (status === 'SUCCESS' && data) { _updateStatus(data); _hideAllBanners(); }
            else _showError(error?.message || 'No se pudo obtener el estado.');
            break;

          case MESSAGE_TYPES.CLOCK_RES:
            _setLoading(btnEntrada, false); _setLoading(btnSalida, false);
            _setLoading(btnPausaInicio, false); _setLoading(btnPausaFin, false);
            if (status === 'SUCCESS') {
              _showInfo('Fichaje registrado correctamente.');
              setTimeout(() => _sendToVelo(MESSAGE_TYPES.GET_STATE, {}), 500);
            } else _showError(error?.message || 'Error al registrar el fichaje.');
            break;

          case MESSAGE_TYPES.GET_HISTORY_RES:
            if (status === 'SUCCESS') { _renderHistory(data || []); _hideAllBanners(); }
            else _showError(error?.message || 'Error al cargar el historial.');
            break;

          case MESSAGE_TYPES.HOURS_CALC_RES:
            _setLoading(btnCalcHours, false);
            if (status === 'SUCCESS') { _renderHoursResult(data); _hideAllBanners(); }
            else _showError(error?.message || 'Error al calcular horas.');
            break;

          case MESSAGE_TYPES.ADJUSTMENT_RES:
            _setLoading(btnAdjSubmit, false);
            if (status === 'SUCCESS') {
              _showInfo('Ajuste registrado correctamente.');
              adjMotivo.value = ''; adjDateTime.value = '';
              setTimeout(() => _sendToVelo(MESSAGE_TYPES.GET_STATE, {}), 500);
            } else _showError(error?.message || 'Error al registrar el ajuste.');
            break;

          case MESSAGE_TYPES.CASHIER_STATE_RES:
            if (status === 'SUCCESS' && data) _renderCashierState(data);
            break;

          case MESSAGE_TYPES.TPV_TX_RES:
            _setLoading(btnSupplierPayment, false);
            _setLoading(btnWithdrawal, false);
            if (status === 'SUCCESS') {
              _showInfo('Movimiento de caja registrado correctamente.');
              if (data?.ticket) _renderTicket(data.ticket);
              supplierName.value = ''; supplierNif.value = ''; supplierAmount.value = '';
              supplierConcept.value = ''; supplierInvoiceRef.value = '';
              withdrawalAmount.value = ''; withdrawalConcept.value = '';
              _sendToVelo(MESSAGE_TYPES.CASHIER_STATE, {});
            } else _showError(error?.message || 'Error al registrar el movimiento de caja.');
            break;

          case MESSAGE_TYPES.X_COUNT_RES:
            _setLoading(btnXCount, false);
            if (status === 'SUCCESS') {
              _showInfo('Ticket X generado correctamente.');
              if (data?.ticket) _renderTicket(data.ticket);
            } else _showError(error?.message || 'Error al generar Ticket X.');
            break;

          case MESSAGE_TYPES.Z_CLOSING_RES:
            _setLoading(btnZClosing, false);
            if (status === 'SUCCESS') {
              _showInfo('Cierre Z Diario completado.');
              if (data?.ticket) _renderTicket(data.ticket);
              _sendToVelo(MESSAGE_TYPES.CASHIER_STATE, {});
            } else _showError(error?.message || 'Error en el Cierre Z Diario.');
            break;

          case MESSAGE_TYPES.ONLY_STAFF_ERROR:
          case MESSAGE_TYPES.STAFF_ERROR:
            _showError(error?.message || 'Error en la solicitud.');
            _setLoading(btnCalcHours, false); _setLoading(btnAdjSubmit, false);
            _setLoading(btnSupplierPayment, false); _setLoading(btnWithdrawal, false);
            _setLoading(btnXCount, false); _setLoading(btnZClosing, false);
            break;
        }
      }

      window.addEventListener('message', (e) => {
        const msg = e?.data || {};
        const type = String(msg.type || '').trim().toUpperCase();
        if (!type || !VALID_TYPES.has(type)) return;
        if (e.source !== window.parent) return;
        if (e.origin && e.origin !== 'null') PARENT_ORIGIN = e.origin;
        if (type === MESSAGE_TYPES.READY) return;
        if (type === MESSAGE_TYPES.CONTEXT) { _renderContext(msg.payload || {}); return; }
        _handleResponse(msg);
      });

      /* ═══════════ Handshake + fallback ═══════════ */
      _sendToVelo(MESSAGE_TYPES.READY, { traceId });
      setTimeout(() => { if (!contextData) _sendToVelo(MESSAGE_TYPES.READY, { traceId }); }, 600);
      setTimeout(() => { if (!contextData) _sendToVelo(MESSAGE_TYPES.READY, { traceId }); }, 1800);

      /* Fallback: si a los 3 s no llegó MM_CONTEXT, aplicamos defaults
         equivalentes a lo que envía zonastaff.js v5004. Evita pantalla
         en blanco. Cuando llegue el contexto real, sobreescribirá. */
      contextFallbackTimer = setTimeout(() => {
        if (!contextData) {
          console.warn('[only-staff] MM_CONTEXT no recibido, aplicando defaults');
          // Fallback fail-closed: sin contexto de autorización no se
          // exponen controles de jornada, caja, inventario ni administración.
          _renderContext({
            isAuthorized: false,
            panelType: 'TPV',
            enableClocking: false,
            enableBreaks: false,
            enableCashier: false,
            enableScheduleQuery: false,
            isAdmin: false,
            isCajero: false,
            cashierMode: 'DISABLED',
            currencyCode: 'EUR',
            timeZone: 'Europe/Madrid'
          });
        }
      }, 3000);
    })();
  </script>
