// SEC-01: origen del padre calculado en tiempo de ejecución. Vive FUERA

const clean = (value, fallback = "") => String(value ?? "").trim() || fallback;
const number = (value, fallback = 0) => {
  if (value === null || value === undefined || value === "") return fallback;
  const parsed = Number(value);
  return Number.isFinite(parsed) ? parsed : fallback;
};

 * SEC-01: utilidades de origen

function normalizeOrigin(value) {
  const raw = clean(value).toLowerCase();

  if (!raw || raw === "null") {
    return "";
  }

  try {
    const url = new URL(raw);

    if (url.protocol !== "https:") {
      return "";
    }

    return url.origin.toLowerCase();
  } catch (_) {
    return "";
  }
}

function hostMatchesSuffix(host, suffix) {
  if (!host || !suffix) {
    return false;
  }

  const normalizedSuffix = suffix
    .replace(/^\.+/, "")
    .toLowerCase();

  return (
    host === normalizedSuffix ||
    host.endsWith(`.${normalizedSuffix}`)
  );
}

function isTrustedOrigin(origin) {
  const normalizedOrigin = normalizeOrigin(origin);

  if (!normalizedOrigin) {
    return false;
  }

  try {
    const host = new URL(normalizedOrigin).hostname.toLowerCase();

    return CONFIG.ALLOWED_ORIGIN_SUFFIXES.some((suffix) =>
      hostMatchesSuffix(host, suffix)
    );
  } catch (_) {
    return false;
  }
}

function resolveParentOrigin() {
  try {
    if (
      window.location.ancestorOrigins &&
      window.location.ancestorOrigins.length
    ) {
      const origin = normalizeOrigin(
        window.location.ancestorOrigins[0]
      );

      if (origin && isTrustedOrigin(origin)) {
        return origin;
      }
    }
  } catch (_) {
    // Continuar con document.referrer.
  }

  try {
    const origin = normalizeOrigin(document.referrer);

    if (origin && isTrustedOrigin(origin)) {
      return origin;
    }
  } catch (_) {
    // Origen no válido.
  }

  return "";
}

function refId(value) {
  if (typeof value === "string" || typeof value === "number") return clean(value);
  if (!value || typeof value !== "object") return "";
  return clean(value.addonId || value._id || value.id || value.referenceId || value.value);
}

// Devuelve el primer identificador válido de la lista de candidatos.
// Importante: se evalúa cada candidato por separado, para que un objeto
// Wix sin claves reconocidas no impida probar los siguientes candidatos.
function firstRefId() {
  for (let i = 0; i < arguments.length; i += 1) {
    const resolved = refId(arguments[i]);
    if (resolved) return resolved;
  }
  return "";
}

function showError(message) {
  const banner = el("errorBanner");
  if (!banner) return;
  banner.textContent = clean(message, "Se produjo un error.");
  banner.style.display = "block";
}

function hideError() {
  const banner = el("errorBanner");
  if (!banner) return;
  banner.textContent = "";
  banner.style.display = "none";
}

function formatPrice(value) {
  try {
    return new Intl.NumberFormat("es-ES", {
      style: "currency", currency: state.currency, minimumFractionDigits: 2, maximumFractionDigits: 2
    }).format(number(value));
  } catch (_) {
    return `${number(value).toFixed(2).replace(".", ",")} ${state.currency}`;
  }
}

function normalizeImage(value) {
  let image = "";
  if (typeof value === "string") image = value.trim();
  else if (value && typeof value === "object") {
    image = clean(value.url || value.src || value.fileUrl || value.uri || value.imageUrl);
  }
  if (!image) return CONFIG.DEFAULT_IMAGE;

  // Las colecciones Wix suelen devolver wix:image://v1/{mediaId}/...; convertirlo a CDN.
  if (/^wix:image:\/\/v1\//i.test(image)) {
    const path = image.replace(/^wix:image:\/\/v1\//i, "").split(/[?#]/, 1)[0];
    const mediaId = path.split("/")[0];
    if (mediaId && /^[a-z0-9._~%-]+$/i.test(mediaId)) {
      return `https://static.wixstatic.com/media/${mediaId}`;
    }
    return CONFIG.DEFAULT_IMAGE;
  }
  if (/^https:\/\/static\.(wixstatic|parastorage)\.com\//i.test(image)) return image;
  if (/^data:image\/(svg\+xml|png|jpe?g|webp|gif);/i.test(image)) return image;
  return CONFIG.DEFAULT_IMAGE;
}

function applyImage(value, title) {
  const image = el("serviceImage");
  if (!image) return;
  image.alt = `Imagen de ${clean(title, "servicio")}`;
  image.onerror = () => {
    image.onerror = null;
    image.src = CONFIG.DEFAULT_IMAGE;
  };
  image.src = normalizeImage(value);
}

function normalizeAddOn(addon) {
  if (!addon || typeof addon !== "object") return null;
  const addonId = refId(addon.addonId || addon._id || addon.id);
  if (!addonId) return null;
  // PATCH C: ComplementosCatalogo.description es RICH_TEXT en CMS.
  // Strip defensivo de tags bien formados (<p>, </p>, <br/>, <!---->) y
  // colapso de espacios residuales. El regex excluye comparaciones
  // literales tipo "5 < 10 y > 3" (no empiezan por letra, "/" ni "!").
  const rawDescription = clean(addon.description);
  const cleanDescription = rawDescription
    .replace(/<[a-zA-Z\/!][^>]*>/g, " ")
    .replace(/\s+/g, " ")
    .trim();
  return {
    addonId,
    name: clean(addon.name || addon.nombreComplemento, "Complemento"),
    description: cleanDescription,
    durationMinutes: Math.max(0, number(addon.durationMinutes)),
    price: Math.max(0, number(addon.price)),
    taxIncluded: addon.taxIncluded === true,
    tipoImpositivo: number(addon.tipoImpositivo)
  };
}

function normalizeRecommendation(value) {
  if (value && typeof value === "object") {
    return clean(value.name || value.title || value.text || value.description, "Producto recomendado");
  }
  return clean(value, "Producto recomendado");
}

function updatePrice() {
  const extras = state.selectedAddOns.reduce((total, addon) => total + addon.price, 0);
  el("servicePrice").textContent = formatPrice(state.price + extras);
  const summary = el("extrasSummary");
  if (!state.selectedAddOns.length) {
    summary.textContent = "";
    summary.style.display = "none";
    return;
  }
  summary.textContent = `Complementos seleccionados: +${formatPrice(extras)}`;
  summary.style.display = "block";
}

function renderAddOns() {
  const section = el("addonsSection");
  const container = el("addonsContainer");
  container.replaceChildren();
  state.selectedAddonIds.clear();
  state.selectedAddOns = [];
  section.hidden = !state.addOnOptions.length;
  if (section.hidden) {
    updatePrice();
    return;
  }

  state.addOnOptions.forEach((addon) => {
    const row = document.createElement("label");
    const checkbox = document.createElement("input");
    const name = document.createElement("span");
    const price = document.createElement("span");
    row.className = "addon";
    checkbox.type = "checkbox";
    checkbox.value = addon.addonId;
    checkbox.setAttribute("aria-label", `Añadir ${addon.name}`);
    name.className = "addon-name";
    name.textContent = addon.name;
    price.className = "addon-price";
    price.textContent = `+${formatPrice(addon.price)}`;
    checkbox.addEventListener("change", () => {
      if (checkbox.checked) state.selectedAddonIds.add(addon.addonId);
      else state.selectedAddonIds.delete(addon.addonId);
      state.selectedAddOns = state.addOnOptions.filter((item) => state.selectedAddonIds.has(item.addonId));
      updatePrice();
    });
    row.append(checkbox, name, price);
    container.appendChild(row);
  });
}

function renderRecommendations(items) {
  const section = el("recommendationsSection");

  container.replaceChildren();

  recommendations.forEach((item) => {
    const recommendation = document.createElement("div");
    recommendation.className = "recommendation";
    recommendation.textContent = normalizeRecommendation(item);
    container.appendChild(recommendation);
  });
}

function renderService(service) {
  if (!service || typeof service !== "object" || Array.isArray(service)) {
    showError("El servicio recibido no es válido.");
    return false;
  }

  // Trazas para diagnosticar por qué se rechaza un contexto en el editor de Wix.
  console.log("[service-widget] renderService input:", {
    serviceId: service.serviceId,
    id: service.id,
    _id: service._id,
    referenceId: service.referenceId,
    pricingPlanId: service.pricingPlanId,
    catalogEntryType: service.catalogEntryType,
    slug: service.slug,
    serviceSlug: service.serviceSlug,
    urlSlug: service.urlSlug,
    clientHidden: service.clientHidden
  });

  // El backend puede entregar dos tipos de entrada de catálogo:
  //  - SERVICE: servicio normal, con serviceId.
  //  - PRICING_PLAN: plan de precios, con serviceId vacío y pricingPlanId.
  const catalogEntryType = clean(service.catalogEntryType, "SERVICE").toUpperCase();

  // Aceptar los distintos nombres con los que Wix puede exponer los
  // identificadores del servicio y de su slug.
  const serviceId = firstRefId(
    service.serviceId,
    service.id,
    service._id,
    service.referenceId
  );

  const pricingPlanId = firstRefId(
    service.pricingPlanId
  );

  const slug = clean(
    service.slug || service.serviceSlug || service.urlSlug
  );

  // El identificador válido depende del tipo de entrada: los planes
  // deliberadamente no llevan serviceId.
  const validReference =
    catalogEntryType === "PRICING_PLAN"
      ? pricingPlanId
      : serviceId;

  if (!validReference || !slug || service.clientHidden === true) {
    console.warn("[service-widget] contexto rechazado:", {
      catalogEntryType,
      serviceId,
      pricingPlanId,
      slug,
      clientHidden: service.clientHidden
    });
    showError("El servicio no está disponible para reserva.");
    return false;
  }

  const title = clean(service.title, "Servicio");
  const duration = Math.max(0, number(service.totalDuration));
  const currency = clean(service.currency, CONFIG.DEFAULT_CURRENCY).toUpperCase();

  // El DTO público de reservas.web.js devuelve recommendations, no recomendaciones.
  const recommendations = Array.isArray(service.recommendations)
    ? service.recommendations
    : (Array.isArray(service.recomendaciones) ? service.recomendaciones : []);

  state.catalogEntryType = catalogEntryType;
  state.serviceId = serviceId;
  state.pricingPlanId = pricingPlanId;
  state.slug = slug;
  state.currency = /^[A-Z]{3}$/.test(currency) ? currency : CONFIG.DEFAULT_CURRENCY;
  state.price = Math.max(0, number(service.price));
  state.addOnOptions = (Array.isArray(service.addOnOptions) ? service.addOnOptions : [])
    .slice(0, CONFIG.MAX_ADDONS)
    .map(normalizeAddOn)
    .filter(Boolean);

  el("serviceTitle").textContent = title;
  el("serviceSummary").textContent = clean(service.tagLine, "Descubre una experiencia personalizada en Marian Madrid.");
  el("serviceDescription").textContent = clean(service.description, "Detalles del servicio no disponibles.");
  el("serviceLocation").textContent = clean(service.location, CONFIG.DEFAULT_LOCATION);
  const durationLabel = duration > 0 ? `${duration} min` : "Consultar";
  el("serviceDuration").textContent = durationLabel;
  el("bookingDuration").textContent = `Duración aproximada: ${durationLabel}`;

  // El backend lee ServiciosCatalogo.mainMedia y normalmente lo proyecta a URL.
  // normalizeImage también resuelve el formato wix:image://v1/... cuando llega sin convertir.
  applyImage(service.mainMedia, title);
  renderAddOns();
  renderRecommendations(recommendations);
  updatePrice();

  state.contextReceived = true;
  state.bookingStarted = false;
  clearTimeout(contextTimer);
  clearTimeout(handshakeTimer);
  handshakeTimer = null;
  contextTimer = null;

  const button = el("btnBookNow");
  button.disabled = false;
  button.textContent = "Continuar reserva";
  hideError();
  return true;
}

function createMessageId() {
  sequence += 1;
  return `service-${Date.now().toString(36)}-${sequence.toString(36)}`;
}

function sendMessage(type, payload = {}) {
  if (!window.parent || window.parent === window) {
    return false;
  }

  const normalizedType = clean(type).toUpperCase();

  if (!Object.values(MESSAGE_TYPES).includes(normalizedType)) {
    return false;
  }

  if (!parentOrigin) {
    console.warn(
      "[service-widget] Origen del padre desconocido"
    );
    return false;
  }

  try {
    const message = {
      version: CONFIG.PROTOCOL_VERSION,
      type: normalizedType,
      payload: payload && typeof payload === "object"
        ? payload
        : {},
      messageId: createMessageId()
    };

    if (state.bridgeId) {
      message.bridgeId = state.bridgeId;
    }

    window.parent.postMessage(message, parentOrigin);
    return true;
  } catch (error) {
    console.error(
      "[service-widget] Error enviando mensaje",
      error
    );
    return false;
  }
}

function sendReady() {
  return sendMessage(MESSAGE_TYPES.READY, { status: "INIT" });
}

function startHandshake() {
  clearTimeout(handshakeTimer);
  clearTimeout(contextTimer);

  handshakeAttempt = 0;
  state.contextReceived = false;

  function retry() {
    if (
      state.contextReceived ||
      handshakeAttempt >= CONFIG.HANDSHAKE_ATTEMPTS
    ) {
      return;
    }

    handshakeAttempt += 1;
    sendReady();

    const delay = Math.min(
      CONFIG.HANDSHAKE_BASE_BACKOFF_MS *
        2 ** (handshakeAttempt - 1),
      CONFIG.HANDSHAKE_MAX_BACKOFF_MS
    );

    handshakeTimer = setTimeout(retry, delay);
  }

  retry();

  contextTimer = setTimeout(() => {
    if (!state.contextReceived) {
      showError(
        "No llegaron los datos del servicio. Recarga la página."
      );
    }
  }, CONFIG.CONTEXT_TIMEOUT_MS);
}

// Gestiona la respuesta del frontend a un MM_BOOK.
// servicio-2.js SIEMPRE responde (SUCCESS al comprar un plan o al navegar a
// la URL de reserva; ERROR si algo falla, incluido PRICING_PLAN_PURCHASE_FAILED,
// BOOKING_URL_FAILED y SERVICE_NOT_READY). Si no se procesa, el usuario ve el
// mensaje genérico de timeout a los 60 s en lugar del error real del backend.
function handleBookReply(payload) {
  const button = el("btnBookNow");
  const data = (payload && typeof payload === "object") ? payload : {};
  const status = clean(data.status).toUpperCase();
  const error = data.error && typeof data.error === "object" ? data.error : {};

  if (status === "SUCCESS") {
    // El frontend ya está navegando (checkout o wixLocation.to). Mantener el
    // botón bloqueado y el texto "Abriendo calendario..." para no duplicar.
    clearTimeout(bookTimer);
    bookTimer = null;
    return;
  }

  if (status === "ERROR") {
    clearTimeout(bookTimer);
    bookTimer = null;
    state.bookingStarted = false;
    if (button) {
      button.disabled = false;
      button.textContent = "Continuar reserva";
    }
    console.warn("[service-widget] MM_BOOK respondido con error:", error);
    showError(clean(error.message, "No se pudo iniciar la reserva. Inténtalo de nuevo."));
  }
}

function handleIncomingMessage(event) {
  // SEC-01 (P0): validar origin ANTES de procesar nada. Rechaza orígenes
  // opacos, orígenes fuera de la allowlist y, cuando se conoce, cualquier
  // origen distinto del documento padre real.
  const eventOrigin = normalizeOrigin(event?.origin);

  if (
    !event ||
    !eventOrigin ||
    !isTrustedOrigin(eventOrigin) ||
    (parentOrigin && eventOrigin !== parentOrigin)
  ) {
    try {
      console.warn(
        "[service-widget] SEC: mensaje de origen no confiable ignorado",
        { origin: event && event.origin }
      );
    } catch (_) { /* no-op */ }
    return;
  }

  const message = event.data;

  if (
    !message ||
    typeof message !== "object" ||
    Array.isArray(message)
  ) {
    return;
  }

  if (Number(message.version) !== CONFIG.PROTOCOL_VERSION) {
    return;
  }

  const type = clean(message.type).toUpperCase();

  if (type === MESSAGE_TYPES.CONTEXT) {
    console.log("[service-widget] mensaje recibido:", {
      type: message.type,
      version: message.version,
      bridgeId: message.bridgeId,
      payload: message.payload
    });

    // Al recibir un MM_CONTEXT, se detienen los timers de handshake y
    // contexto: ya no tiene sentido seguir esperando ni mostrar el
    // mensaje genérico de timeout.
    clearTimeout(handshakeTimer);
    clearTimeout(contextTimer);
    handshakeTimer = null;
    contextTimer = null;

    state.bridgeId = clean(message.bridgeId);
    const context = message.payload;

    // servicio-2.js puede enviar un MM_CONTEXT con { error: { code, message } }
    // si loadService() falla, para que no se espere al timeout y se muestre
    // el mensaje real de inmediato.
    const ctxError =
      context && typeof context === "object" ? context.error : null;

    if (ctxError && typeof ctxError === "object") {
      console.warn("[service-widget] MM_CONTEXT con error:", ctxError);
      showError(clean(ctxError.message, "No se pudo cargar el servicio."));
      return;
    }

    renderService(context);
    return;
  }

  if (type === MESSAGE_TYPES.BOOK) {
    handleBookReply(message.payload);
    return;
  }
}

function initialize() {
  parentOrigin = resolveParentOrigin();

  if (!parentOrigin) {
    console.warn(
      "[service-widget] No se pudo determinar el origen del padre."
    );
  }

  applyImage("", "servicio");

  el("btnBack").addEventListener("click", () => {
    sendMessage(MESSAGE_TYPES.NAV, {
      target: "SERVICIOS"
    });
  });

  el("btnBookNow").addEventListener("click", () => {
    if (!state.contextReceived) {
      showError(
        "Conectando con la página. Espera un momento."
      );
      return;
    }

    if (state.bookingStarted) {
      return;
    }

    state.bookingStarted = true;

    const button = el("btnBookNow");
    button.disabled = true;
    button.textContent = "Abriendo calendario...";
    hideError();

    const bookPayload = {
      serviceId: state.serviceId,
      slug: state.slug,
      addOnIds: Array.from(state.selectedAddonIds)
        .slice(0, CONFIG.MAX_ADDONS)
    };

    if (
      state.catalogEntryType === "PRICING_PLAN" &&
      state.pricingPlanId
    ) {
      bookPayload.pricingPlanId = state.pricingPlanId;
    }

    if (!sendMessage(MESSAGE_TYPES.BOOK, bookPayload)) {
      state.bookingStarted = false;
      button.disabled = false;
      button.textContent = "Continuar reserva";
      showError(
        "No se pudo iniciar la reserva. Recarga la página."
      );
      return;
    }

    clearTimeout(bookTimer);

    bookTimer = setTimeout(() => {
      if (
        state.bookingStarted &&
        document.visibilityState === "visible"
      ) {
        showError(
          "La reserva sigue pendiente. No pulses de nuevo."
        );
      }
    }, CONFIG.BOOK_NOTICE_TIMEOUT_MS);
  });

  window.addEventListener(
    "message",
    handleIncomingMessage
  );

  startHandshake();
}

if (document.readyState === "loading") {
  document.addEventListener("DOMContentLoaded", initialize, { once: true });
} else {
  initialize();
}
